def load_json(path: str, default):
    try:
        if os.path.exists(path):
            with open(path, "r", encoding="utf-8") as f:
                return json.load(f)
    except Exception:
        pass
    return default


def save_json(path: str, data):
    with open(path, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=2)

def build_stats_embed(member: discord.Member, stats: dict) -> discord.Embed:
    attended = int(stats.get("attended", 0))
    hosted = int(stats.get("hosted", 0))
    no_shows = int(stats.get("no_shows", 0))
    warnings = int(stats.get("warnings", 0))
    strikes = int(stats.get("strikes", 0))

    score = max(0, (attended * 5) + (hosted * 8) - (no_shows * 6) - (warnings * 3) - (strikes * 10))
    if score >= 80:
        grade = "A"
    elif score >= 60:
        grade = "B"
    elif score >= 40:
        grade = "C"
    else:
        grade = "D"

    embed = discord.Embed(
        title=f"📈 {member.display_name} — My DIFF Stats",
        description="Your current DIFF activity snapshot from the leaderboard system.",
        color=discord.Color.green()
    )
    embed.add_field(name="✅ Meets Attended", value=str(attended), inline=True)
    embed.add_field(name="🎤 Meets Hosted", value=str(hosted), inline=True)
    embed.add_field(name="❌ No-Shows", value=str(no_shows), inline=True)
    embed.add_field(name="⚠️ Warnings", value=str(warnings), inline=True)
    embed.add_field(name="🚨 Strikes", value=str(strikes), inline=True)
    embed.add_field(name="🏅 Activity Score", value=f"{score} ({grade})", inline=True)
    embed.set_footer(text="Stats are pulled from diff_activity_data.json")
    return embed

class DiffControlHubView(discord.ui.View):
    def __init__(self):
        super().__init__(timeout=None)

    @discord.ui.button(label="📝 Crew Roll Call", style=discord.ButtonStyle.link, url=ROLL_CALL_CHANNEL, row=0)
    async def rollcall(self, interaction: discord.Interaction, button: discord.ui.Button):
        pass

    @discord.ui.button(label="🎨 Crew Color Voting", style=discord.ButtonStyle.link, url=COLOR_CHANNEL, row=0)
    async def color(self, interaction: discord.Interaction, button: discord.ui.Button):
        pass

    @discord.ui.button(label="⚠️ Strike System", style=discord.ButtonStyle.primary, row=1)
    async def strike(self, interaction: discord.Interaction, button: discord.ui.Button):
        await interaction.response.send_message(embed=build_strike_embed(), ephemeral=True)

    @discord.ui.button(label="🧥 Crew Jackets", style=discord.ButtonStyle.secondary, row=1)
    async def jackets(self, interaction: discord.Interaction, button: discord.ui.Button):
        await interaction.response.send_message(embed=build_jackets_embed(), ephemeral=True)

    @discord.ui.button(label="📋 Roles & Responsibility", style=discord.ButtonStyle.secondary, row=2)
    async def roles(self, interaction: discord.Interaction, button: discord.ui.Button):
        await interaction.response.send_message(embed=build_roles_embed(), ephemeral=True)

    @discord.ui.button(label="📈 My Stats", style=discord.ButtonStyle.success, row=2)
    async def stats(self, interaction: discord.Interaction, button: discord.ui.Button):
        data = load_json(STATS_FILE, {})
        member_stats = data.get(str(interaction.user.id), {})
        embed = build_stats_embed(interaction.user, member_stats)
        await interaction.response.send_message(embed=embed, ephemeral=True)


class DiffControlHubConnected(commands.Cog):
    def __init__(self, bot):
        self.bot = bot
        self.bot.add_view(DiffControlHubView())

    async def post_or_refresh_hub(self, channel: discord.abc.Messageable, guild: discord.Guild):
        state = load_json(STATE_FILE, {})
        guild_key = str(guild.id)
        guild_state = state.get(guild_key, {})
        channel_id = guild_state.get("channel_id")
        message_id = guild_state.get("message_id")

        embed = build_hub_embed()
        view = DiffControlHubView()

        existing_message = None
        if channel_id and message_id:
            saved_channel = guild.get_channel(int(channel_id))
            if saved_channel:
                try:
                    existing_message = await saved_channel.fetch_message(int(message_id))
                except Exception:
                    existing_message = None

        if existing_message:
            await existing_message.edit(embed=embed, view=view)
            return existing_message, False

        posted = await channel.send(embed=embed, view=view)
        state[guild_key] = {"channel_id": str(channel.id), "message_id": str(posted.id)}
        save_json(STATE_FILE, state)
        return posted, True

    @app_commands.command(name="diffhub-connected", description="Post or refresh the DIFF Crew Control Hub")
    async def diffhub_connected(self, interaction: discord.Interaction):
        _, created = await self.post_or_refresh_hub(interaction.channel, interaction.guild)
        msg = "DIFF Control Hub posted ✅" if created else "DIFF Control Hub refreshed ✅"
        await interaction.response.send_message(msg, ephemeral=True)

    @app_commands.command(name="diffhub-refresh", description="Refresh the saved DIFF Crew Control Hub")
    async def diffhub_refresh(self, interaction: discord.Interaction):
        await self.post_or_refresh_hub(interaction.channel, interaction.guild)
        await interaction.response.send_message("DIFF Control Hub refreshed ✅", ephemeral=True)


async def setup(bot):
    await bot.add_cog(DiffControlHubConnected(bot))
