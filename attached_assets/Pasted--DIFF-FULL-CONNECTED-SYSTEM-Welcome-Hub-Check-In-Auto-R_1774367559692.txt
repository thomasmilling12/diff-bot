# DIFF FULL CONNECTED SYSTEM
# Welcome Hub + Check-In + Auto Role System

import discord
from discord.ext import commands

GUILD_ID = 850386896509337710

WELCOME_CHANNEL_ID = 1486006000808103986
RULES_CHANNEL_ID = 0
JOIN_MEETS_CHANNEL_ID = 1277084633858576406

VERIFIED_ROLE_ID = 1141424243616256032
UNVERIFIED_ROLE_ID = 0

class DiffSystem(commands.Cog):
    def __init__(self, bot):
        self.bot = bot

    @commands.Cog.listener()
    async def on_member_join(self, member):
        if member.guild.id != GUILD_ID:
            return

        # Give unverified role
        if UNVERIFIED_ROLE_ID != 0:
            role = member.guild.get_role(UNVERIFIED_ROLE_ID)
            if role:
                await member.add_roles(role)

        channel = self.bot.get_channel(WELCOME_CHANNEL_ID)

        embed = discord.Embed(
            title="🏁 DIFF CHECK-IN REQUIRED",
            description=(
                f"{member.mention}\n\n"
                "Before entering the meet area, complete:\n\n"
                f"• Go to <#{RULES_CHANNEL_ID}> and get verified\n"
                f"• Verify in <#{JOIN_MEETS_CHANNEL_ID}>\n"
                "• Get cleared to enter\n\n"
                "🔒 Access is locked until verified"
            ),
            color=discord.Color.blue()
        )

        view = discord.ui.View()

        view.add_item(discord.ui.Button(
            label="Go to Rules",
            url=f"https://discord.com/channels/{GUILD_ID}/{RULES_CHANNEL_ID}"
        ))

        view.add_item(discord.ui.Button(
            label="Go to Join Meets",
            url=f"https://discord.com/channels/{GUILD_ID}/{JOIN_MEETS_CHANNEL_ID}"
        ))

        await channel.send(embed=embed, view=view)

    @commands.Cog.listener()
    async def on_member_update(self, before, after):
        if after.guild.id != GUILD_ID:
            return

        before_roles = [r.id for r in before.roles]
        after_roles = [r.id for r in after.roles]

        # If user gets verified role
        if VERIFIED_ROLE_ID in after_roles and VERIFIED_ROLE_ID not in before_roles:

            # Remove unverified
            if UNVERIFIED_ROLE_ID != 0:
                role = after.guild.get_role(UNVERIFIED_ROLE_ID)
                if role and role in after.roles:
                    await after.remove_roles(role)

            try:
                await after.send(
                    "✅ You are now verified in DIFF. You have full access."
                )
            except:
                pass

async def setup(bot):
    await bot.add_cog(DiffSystem(bot))
