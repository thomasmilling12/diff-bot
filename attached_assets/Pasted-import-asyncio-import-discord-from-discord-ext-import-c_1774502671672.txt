import asyncio
import discord
from discord.ext import commands
from discord import app_commands

# =========================================================
# DIFF Upcoming Meet Info Panel
# ---------------------------------------------------------
# Posts a clean info panel explaining what this channel is for
# in channel: 1485861257708834836
#
# Features:
# - Auto-posts on bot startup
# - Removes old bot-made copies first (anti-dupe)
# - Adds link buttons to important meet channels
# - Includes slash command to refresh the panel
# =========================================================

UPCOMING_MEET_CHANNEL_ID = 1485861257708834836
STAFF_ROLE_ID = 123456789012345678  # CHANGE THIS

PANEL_TITLE = "📅 DIFF Upcoming Meet Hub"
PANEL_DESCRIPTION = (
    "**What this channel is used for**\n\n"
    "This channel is used to keep members updated on **upcoming DIFF meets** before they begin.\n\n"
    "You can use this channel to:\n"
    "• Check what meet is coming up next\n"
    "• See official meet posts and timing\n"
    "• Stay ready for future public or pop-up meet drops\n"
    "• Know where to go once a meet announcement is live\n\n"
    "━━━━━━━━━━━━━━━━━━━━━━\n"
    "**Quick Access Buttons Below**\n"
    "Use the buttons to jump straight to important meet channels.\n"
    "━━━━━━━━━━━━━━━━━━━━━━"
)

BUTTONS = [
    ("🏁 Official Meet Posts", "https://discord.com/channels/850386896509337710/1485870611069796374"),
    ("⚡ Pop-Up Meets", "https://discord.com/channels/850386896509337710/1484768466023223418"),
    ("💬 Meet Chat", "https://discord.com/channels/850386896509337710/1195953265377021952"),
    ("📸 Meet Media", "https://discord.com/channels/850386896509337710/1266933655486332999"),
]


def is_staff_member(user: discord.abc.User) -> bool:
    if not isinstance(user, discord.Member):
        return False
    return any(role.id == STAFF_ROLE_ID for role in user.roles) or user.guild_permissions.administrator


class UpcomingMeetPanelView(discord.ui.View):
    def __init__(self):
        super().__init__(timeout=None)

        for label, url in BUTTONS:
            self.add_item(
                discord.ui.Button(
                    label=label,
                    url=url,
                    style=discord.ButtonStyle.link
                )
            )


class UpcomingMeetPanelCog(commands.Cog):
    def __init__(self, bot: commands.Bot):
        self.bot = bot
        self.view = UpcomingMeetPanelView()

    async def post_or_refresh_panel(self):
        await self.bot.wait_until_ready()
        await asyncio.sleep(2)

        channel = self.bot.get_channel(UPCOMING_MEET_CHANNEL_ID)
        if channel is None or not isinstance(channel, discord.TextChannel):
            print("Upcoming meet panel channel not found.")
            return

        try:
            async for msg in channel.history(limit=50):
                if msg.author.id != self.bot.user.id:
                    continue

                panel_match = False
                if msg.embeds:
                    for emb in msg.embeds:
                        if emb.title == PANEL_TITLE:
                            panel_match = True
                            break

                button_match = False
                for row in msg.components:
                    for component in row.children:
                        if getattr(component, "url", None) in [url for _, url in BUTTONS]:
                            button_match = True
                            break

                if panel_match or button_match:
                    await msg.delete()

        except discord.Forbidden:
            print("Missing permission to delete old panel messages.")
        except Exception as e:
            print(f"Error cleaning old upcoming meet panels: {e}")

        embed = discord.Embed(
            title=PANEL_TITLE,
            description=PANEL_DESCRIPTION,
            color=discord.Color.gold(),
        )
        embed.set_footer(text="DIFF Meets • Upcoming Meet Information")

        await channel.send(embed=embed, view=self.view)

    @commands.Cog.listener()
    async def on_ready(self):
        self.bot.add_view(self.view)
        try:
            await self.post_or_refresh_panel()
        except Exception as e:
            print(f"Failed to post upcoming meet panel: {e}")

    @app_commands.command(
        name="refresh_upcoming_meet_panel",
        description="Refresh the upcoming meet info panel."
    )
    async def refresh_upcoming_meet_panel(self, interaction: discord.Interaction):
        if not is_staff_member(interaction.user):
            await interaction.response.send_message(
                "Only staff can refresh this panel.",
                ephemeral=True
            )
            return

        await self.post_or_refresh_panel()
        await interaction.response.send_message(
            "Upcoming meet panel refreshed.",
            ephemeral=True
        )


async def setup(bot: commands.Bot):
    await bot.add_cog(UpcomingMeetPanelCog(bot))
