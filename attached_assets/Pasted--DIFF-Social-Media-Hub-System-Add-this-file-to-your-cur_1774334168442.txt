# =========================
# WELCOME HUB BUTTON VIEW
# =========================
class WelcomeHubSocialButtonView(discord.ui.View):
    def __init__(self) -> None:
        super().__init__(timeout=None)

    @discord.ui.button(
        label="📲 Social Media",
        style=discord.ButtonStyle.secondary,
        custom_id="diff:welcome:social_media",
    )
    async def social_media_button(
        self,
        interaction: discord.Interaction,
        button: discord.ui.Button,
    ) -> None:
        embed = build_social_media_embed()
        view = SocialMediaLinksView()
        await interaction.response.send_message(embed=embed, view=view, ephemeral=True)


# =========================
# COG
# =========================
class SocialMediaHubCog(commands.Cog):
    def __init__(self, bot: commands.Bot) -> None:
        self.bot = bot

    @commands.hybrid_command(name="post_social_hub", description="Post the DIFF social media hub.")
    @commands.has_permissions(manage_guild=True)
    async def post_social_hub(self, ctx: commands.Context, channel: discord.TextChannel | None = None) -> None:
        target = channel or ctx.channel
        if target is None:
            await ctx.reply("I could not find a channel to post in.")
            return

        embed = build_social_media_embed()
        view = SocialMediaLinksView()
        await target.send(embed=embed, view=view)

        if ctx.channel and ctx.channel.id != target.id:
            try:
                await ctx.reply(f"Posted the DIFF Social Media Hub in {target.mention}.")
            except discord.HTTPException:
                pass

    @commands.hybrid_command(name="post_social_button", description="Post the welcome hub social media button.")
    @commands.has_permissions(manage_guild=True)
    async def post_social_button(self, ctx: commands.Context, channel: discord.TextChannel | None = None) -> None:
        target = channel or ctx.channel
        if target is None:
            await ctx.reply("I could not find a channel to post in.")
            return

        embed = discord.Embed(
            title="🌐 DIFF Media Access",
            description=(
                "Press the button below to open the **DIFF Social Media Hub**.\n"
                "Stay connected, catch recent content, and see where your car might get featured."
            ),
            color=discord.Color.dark_grey(),
        )
        embed.set_footer(text="Different Meets • Official Media Access")

        await target.send(embed=embed, view=WelcomeHubSocialButtonView())

        if ctx.channel and ctx.channel.id != target.id:
            try:
                await ctx.reply(f"Posted the social media button in {target.mention}.")
            except discord.HTTPException:
                pass


# =========================
# SETUP
# =========================
async def setup(bot: commands.Bot) -> None:
    await bot.add_cog(SocialMediaHubCog(bot))

    # Persistent views for buttons
    bot.add_view(SocialMediaLinksView())
    bot.add_view(WelcomeHubSocialButtonView())


async def setup_social_media_hub(bot: commands.Bot) -> None:
    """Optional helper if you prefer importing a custom setup function."""
    await setup(bot)
