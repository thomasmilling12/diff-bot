# HELPERS

def load_json(path: Path, default):
    if not path.exists():
        save_json(path, default)
        return default
    with path.open("r", encoding="utf-8") as f:
        return json.load(f)

def save_json(path: Path, data):
    with path.open("w", encoding="utf-8") as f:
        json.dump(data, f, indent=4, ensure_ascii=False)

def ensure_files():
    load_json(STATE_FILE, {"channel_id": MANAGER_HUB_CHANNEL_ID, "message_id": None})
    load_json(STATS_FILE, {"managers": {}})

def blank_stats(member_id: int, name: str) -> dict:
    return {
        "member_id": member_id,
        "name": name,
        "recruitment": 0,
        "meet_support": 0,
        "manager_actions": 0,
        "issues_reported": 0,
        "warnings_filed": 0,
    }

def compute_score(stats: dict) -> int:
    return (
        stats.get("recruitment", 0) * RECRUITMENT_POINTS
        + stats.get("meet_support", 0) * MEET_SUPPORT_POINTS
        + stats.get("manager_actions", 0) * MANAGER_ACTION_POINTS
        + stats.get("issues_reported", 0) * ISSUE_REPORT_POINTS
        + stats.get("warnings_filed", 0) * WARNING_FILED_POINTS
    )

def sort_managers(managers: dict) -> list:
    values = list(managers.values())
    for item in values:
        item["score"] = compute_score(item)
    values.sort(key=lambda x: (x["score"], x.get("recruitment", 0), x.get("meet_support", 0)), reverse=True)
    return values

class InteractiveManagerHub(commands.Cog):
    def __init__(self, bot: commands.Bot):
        self.bot = bot
        ensure_files()
        self.bot.add_view(ManagerHubView(self))
        self.bot.add_view(ManagerActionsView(self))

    # -------------------------
    # Stats
    # -------------------------

    def get_stats_data(self) -> dict:
        return load_json(STATS_FILE, {"managers": {}})

    def save_stats_data(self, data: dict):
        save_json(STATS_FILE, data)

    async def adjust_stats(
        self,
        member: discord.Member,
        recruitment: int = 0,
        meet_support: int = 0,
        manager_actions: int = 0,
        issues_reported: int = 0,
        warnings_filed: int = 0,
    ) -> dict:
        data = self.get_stats_data()
        key = str(member.id)

        if key not in data["managers"]:
            data["managers"][key] = blank_stats(member.id, str(member))

        stats = data["managers"][key]
        stats["name"] = str(member)
        stats["recruitment"] = max(0, stats.get("recruitment", 0) + recruitment)
        stats["meet_support"] = max(0, stats.get("meet_support", 0) + meet_support)
        stats["manager_actions"] = max(0, stats.get("manager_actions", 0) + manager_actions)
        stats["issues_reported"] = max(0, stats.get("issues_reported", 0) + issues_reported)
        stats["warnings_filed"] = max(0, stats.get("warnings_filed", 0) + warnings_filed)

        self.save_stats_data(data)
        await self.post_or_refresh_panel()

        if MANAGER_LOG_CHANNEL_ID:
            log_channel = self.bot.get_channel(MANAGER_LOG_CHANNEL_ID)
            if log_channel:
                changes = []
                if recruitment:
                    changes.append(f"Recruitment {recruitment:+d}")
                if meet_support:
                    changes.append(f"Meet Support {meet_support:+d}")
                if manager_actions:
                    changes.append(f"Manager Actions {manager_actions:+d}")
                if issues_reported:
                    changes.append(f"Issues Reported {issues_reported:+d}")
                if warnings_filed:
                    changes.append(f"Warnings Filed {warnings_filed:+d}")
                if changes:
                    await log_channel.send(f"📈 {member.mention} updated stats: " + " | ".join(changes))

        return stats

    # Embeds
    # -------------------------

            name="Hub Sections",

                "• Crew Recruitment Roles\n"
                "• Discord Manager Roles\n"
                "• PlayStation Group Chat Manager Roles\n"
                "• Problem To Look Out For\n"
                "• Live Manager Leaderboard\n"
                "• My Performance\n"
                "• Manager Actions"
            ),
            inline=False,
        )
        embed.add_field(
            name="How To Use",
            value="Use the dropdown below to open a section. The Manager Actions option opens interactive buttons.",
            inline=False,
        )
        embed.set_footer(text=FOOTER_TEXT)
        if LOGO_URL:
            embed.set_thumbnail(url=LOGO_URL)
        if BANNER_URL:
            embed.set_image(url=BANNER_URL)
        return embed

    def build_leaderboard_embed(self) -> discord.Embed:
        data = self.get_stats_data()
        ranked = sort_managers(data.get("managers", {}))

        embed = discord.Embed(
            title="📈 DIFF Manager Leaderboard",
            description="Live manager rankings based on tracked activity.",
            color=discord.Color.green(),
        )

        if ranked:
            medals = ["🥇", "🥈", "🥉"]
            lines = []
            for idx, item in enumerate(ranked[:10], start=1):
                prefix = medals[idx - 1] if idx <= 3 else f"**#{idx}**"
                lines.append(
                    f"{prefix} <@{item['member_id']}> — **{item['score']} pts**"
                    f" | R:{item['recruitment']} MS:{item['meet_support']} A:{item['manager_actions']}"
                    f" I:{item['issues_reported']} W:{item['warnings_filed']}"
                )
            embed.add_field(name="Top Managers", value="\n".join(lines), inline=False)
        else:
            embed.add_field(name="Top Managers", value="No manager stats have been recorded yet.", inline=False)

        embed.add_field(
            name="Scoring",
            value=(
                f"Recruitment = {RECRUITMENT_POINTS} pts\n"
                f"Meet Support = {MEET_SUPPORT_POINTS} pts\n"
                f"Manager Actions = {MANAGER_ACTION_POINTS} pts\n"
                f"Issues Reported = {ISSUE_REPORT_POINTS} pt\n"
                f"Warnings Filed = {WARNING_FILED_POINTS} pt"
            ),
            inline=False,
        )
        embed.set_footer(text=FOOTER_TEXT)
        if LOGO_URL:
            embed.set_thumbnail(url=LOGO_URL)
        return embed

    def build_profile_embed(self, member: discord.Member) -> discord.Embed:
        data = self.get_stats_data()
        key = str(member.id)
        stats = data["managers"].get(key, blank_stats(member.id, str(member)))
        score = compute_score(stats)

        embed = discord.Embed(
            title=f"📊 My Performance • {member.display_name}",
            description="Your current manager activity snapshot.",
            color=discord.Color.gold(),
        )
        embed.add_field(name="Recruitment", value=str(stats.get("recruitment", 0)), inline=True)
        embed.add_field(name="Meet Support", value=str(stats.get("meet_support", 0)), inline=True)
        embed.add_field(name="Manager Actions", value=str(stats.get("manager_actions", 0)), inline=True)
        embed.add_field(name="Issues Reported", value=str(stats.get("issues_reported", 0)), inline=True)
        embed.add_field(name="Warnings Filed", value=str(stats.get("warnings_filed", 0)), inline=True)
        embed.add_field(name="Total Score", value=str(score), inline=True)
        embed.set_footer(text=FOOTER_TEXT)
        if LOGO_URL:
            embed.set_thumbnail(url=LOGO_URL)
        return embed

    def build_section_embed(self, key: str) -> discord.Embed:
        section = SECTION_CONTENT[key]
        embed = discord.Embed(
            title=f"{section['emoji']} {section['label']}",
            description=section["description"],
            color=discord.Color.blurple(),
        )
        embed.set_footer(text=FOOTER_TEXT)
        if LOGO_URL:
            embed.set_thumbnail(url=LOGO_URL)
        if BANNER_URL:
            embed.set_image(url=BANNER_URL)
        return embed

    def build_actions_embed(self, member: discord.Member) -> discord.Embed:
        stats = self.get_stats_data()["managers"].get(str(member.id), blank_stats(member.id, str(member)))
        embed = discord.Embed(
            title="⚙️ Manager Actions",
            description=(
                "Use the buttons below to update your tracked manager activity.\n\n"
                "Each press updates your score and refreshes the live panel."
            ),
            color=discord.Color.blurple(),
        )
        embed.add_field(name="Recruitment", value=str(stats.get("recruitment", 0)), inline=True)
        embed.add_field(name="Meet Support", value=str(stats.get("meet_support", 0)), inline=True)
        embed.add_field(name="Manager Actions", value=str(stats.get("manager_actions", 0)), inline=True)
        embed.add_field(name="Issues Reported", value=str(stats.get("issues_reported", 0)), inline=True)
        embed.add_field(name="Warnings Filed", value=str(stats.get("warnings_filed", 0)), inline=True)
        embed.add_field(name="Total Score", value=str(compute_score(stats)), inline=True)
        embed.set_footer(text=FOOTER_TEXT)
        return embed

    # -------------------------
    # Panel refresh
    # -------------------------

    async def post_or_refresh_panel(self):
        state = load_json(STATE_FILE, {"channel_id": MANAGER_HUB_CHANNEL_ID, "message_id": None})
        channel_id = state.get("channel_id", MANAGER_HUB_CHANNEL_ID)
        message_id = state.get("message_id")

        channel = self.bot.get_channel(channel_id)
        if channel is None:
            try:
                channel = await self.bot.fetch_channel(channel_id)
            except Exception as e:
                return False, f"Could not access the manager hub channel: {e}"

        embed = self.build_main_embed()
        view = ManagerHubView(self)
        content = f"<@&{MANAGER_ROLE_PING_ID}>" if MANAGER_ROLE_PING_ID else None

        if message_id:
            try:
                msg = await channel.fetch_message(message_id)
                await msg.edit(content=content, embed=embed, view=view)
                return True, "Interactive manager hub refreshed successfully."
            except Exception:
                pass

        try:
            msg = await channel.send(content=content, embed=embed, view=view)
            state["message_id"] = msg.id
            state["channel_id"] = channel.id
            save_json(STATE_FILE, state)
            return True, "Interactive manager hub posted successfully."
        except Exception as e:
            return False, f"Failed to post interactive manager hub: {e}"

class ManagerHubSelect(discord.ui.Select):
    def __init__(self, cog: InteractiveManagerHub):
        self.cog = cog
        options = [
            discord.SelectOption(label=SECTION_CONTENT["crew_manager_roles"]["label"], value="crew_manager_roles", emoji="📋"),
            discord.SelectOption(label=SECTION_CONTENT["crew_recruitment_roles"]["label"], value="crew_recruitment_roles", emoji="📣"),
            discord.SelectOption(label=SECTION_CONTENT["discord_manager_roles"]["label"], value="discord_manager_roles", emoji="🖥️"),
            discord.SelectOption(label=SECTION_CONTENT["ps_group_chat_manager_roles"]["label"], value="ps_group_chat_manager_roles", emoji="🎮"),
            discord.SelectOption(label=SECTION_CONTENT["problems_to_look_out_for"]["label"], value="problems_to_look_out_for", emoji="⛔"),
            discord.SelectOption(label="Live Manager Leaderboard", value="live_leaderboard", emoji="📈"),
            discord.SelectOption(label="My Performance", value="my_performance", emoji="👤"),
            discord.SelectOption(label="Manager Actions", value="manager_actions", emoji="⚙️"),
        ]
        super().__init__(
            placeholder="Select a manager section...",
            min_values=1,
            max_values=1,
            options=options,
            custom_id="diff_interactive_manager_hub_select",
        )

    async def callback(self, interaction: discord.Interaction):
        value = self.values[0]

        if value in SECTION_CONTENT:
            await interaction.response.send_message(embed=self.cog.build_section_embed(value), ephemeral=True)
            return

        if value == "live_leaderboard":
            await interaction.response.send_message(embed=self.cog.build_leaderboard_embed(), ephemeral=True)
            return

        if value == "my_performance":
            if not isinstance(interaction.user, discord.Member):
                await interaction.response.send_message("Could not load your profile.", ephemeral=True)
                return
            await interaction.response.send_message(embed=self.cog.build_profile_embed(interaction.user), ephemeral=True)
            return

        if value == "manager_actions":
            if not isinstance(interaction.user, discord.Member):
                await interaction.response.send_message("Could not open actions.", ephemeral=True)
                return
            await interaction.response.send_message(
                embed=self.cog.build_actions_embed(interaction.user),
                view=ManagerActionsView(self.cog),
                ephemeral=True
            )
            return


class ManagerHubView(discord.ui.View):
    def __init__(self, cog: InteractiveManagerHub):
        super().__init__(timeout=None)
        self.add_item(ManagerHubSelect(cog))


class ManagerActionsView(discord.ui.View):
    def __init__(self, cog: InteractiveManagerHub):
        super().__init__(timeout=None)
        self.cog = cog

    async def _check_perm(self, interaction: discord.Interaction) -> bool:
        if not isinstance(interaction.user, discord.Member):
            await interaction.response.send_message("Could not verify permissions.", ephemeral=True)
            return False
        if not interaction.user.guild_permissions.manage_guild:
            await interaction.response.send_message("You need Manage Server to use manager actions.", ephemeral=True)
            return False
        return True

    async def _refresh_action_response(self, interaction: discord.Interaction):
        member = interaction.user
        embed = self.cog.build_actions_embed(member)
        await interaction.response.edit_message(embed=embed, view=self)

    @discord.ui.button(label="+1 Recruitment", style=discord.ButtonStyle.green, emoji="📣", custom_id="mgr_add_recruitment")
    async def add_recruitment(self, interaction: discord.Interaction, button: discord.ui.Button):

class ManagerHubGroup(app_commands.Group):
    def __init__(self, cog: InteractiveManagerHub):
        super().__init__(name="managerhub", description="Manage the DIFF interactive manager hub")
        self.cog = cog

    async def interaction_check(self, interaction: discord.Interaction) -> bool:
        if not interaction.user.guild_permissions.manage_guild:
            await interaction.response.send_message("You need Manage Server to use this.", ephemeral=True)
            return False
        return True

    @app_commands.command(name="setup", description="Post or refresh the interactive manager hub")
    async def setup_cmd(self, interaction: discord.Interaction):
        ok, msg = await self.cog.post_or_refresh_panel()
        await interaction.response.send_message(msg, ephemeral=True)

    @app_commands.command(name="refresh", description="Refresh the interactive manager hub")
    async def refresh_cmd(self, interaction: discord.Interaction):
        ok, msg = await self.cog.post_or_refresh_panel()
        await interaction.response.send_message(msg, ephemeral=True)

    @app_commands.command(name="leaderboard", description="View the live manager leaderboard")
    async def leaderboard_cmd(self, interaction: discord.Interaction):
        await interaction.response.send_message(embed=self.cog.build_leaderboard_embed(), ephemeral=True)

    @app_commands.command(name="resetall", description="Reset all manager stats")
    async def resetall_cmd(self, interaction: discord.Interaction):
        save_json(STATS_FILE, {"managers": {}})
        await self.cog.post_or_refresh_panel()
        await interaction.response.send_message("All manager stats have been reset.", ephemeral=True)

    await bot.add_cog(cog, guild=discord.Object(id=GUILD_ID))
    bot.tree.add_command(ManagerHubGroup(cog), guild=discord.Object(id=GUILD_ID))
