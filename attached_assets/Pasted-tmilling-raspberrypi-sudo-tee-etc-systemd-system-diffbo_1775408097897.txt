tmilling@raspberrypi:~ $ sudo tee /etc/systemd/system/diffbot.service > /dev/null << 'EOF'
[Unit]
Description=DIFF Discord Bot
After=network-online.target
Wants=network-online.target

[Service]
User=tmilling
WorkingDirectory=/home/tmilling/diffbot
ExecStart=/usr/bin/python3 -u bot.py
Restart=always
RestartSec=10

[Install]
WantedBy=multi-user.target
EOF

sudo systemctl daemon-reload
sudo systemctl enable diffbot
sudo systemctl start diffbot
sudo systemctl status diffbot
