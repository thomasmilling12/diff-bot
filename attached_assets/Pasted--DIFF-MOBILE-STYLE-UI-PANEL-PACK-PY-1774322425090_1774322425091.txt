# ===============================
# DATA HELPERS
# ===============================

def ensure_data_file():
    os.makedirs("./data", exist_ok=True)
    if not os.path.exists(DATA_FILE):
        with open(DATA_FILE, "w", encoding="utf-8") as f:
            json.dump({"users": {}}, f, indent=2)

def load_data():
    ensure_data_file()
    try:
        with open(DATA_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    except Exception:
        return {"users": {}}

def get_top_members(limit=5):
    data = load_data()
    users = []
    for uid, stats in data.get("users", {}).items():
        users.append({
            "user_id": int(uid),
            "attendance": int(stats.get("attendance", 0) or 0),
            "photos": int(stats.get("photos", 0) or 0),
        })
    users.sort(key=lambda x: x["attendance"], reverse=True)
    return users[:limit]

def get_overview():
    data = load_data()
    users = list(data.get("users", {}).values())
    return {
        "tracked": len(users),
        "active": len([u for u in users if int(u.get("attendance", 0) or 0) > 0]),
        "attendance_total": sum(int(u.get("attendance", 0) or 0) for u in users),
        "photo_complete": len([u for u in users if int(u.get("photos", 0) or 0) >= 10]),
    }

class OneButtonRefreshView(discord.ui.View):
    def __init__(self, bot, custom_id, label, emoji, updater):
        super().__init__(timeout=None)
        self.bot = bot
        self.custom_id_value = custom_id
        self.label_value = label
        self.emoji_value = emoji
        self.updater = updater

        self.add_item(
            discord.ui.Button(
                label=self.label_value,
                emoji=self.emoji_value,
                style=discord.ButtonStyle.secondary,
                custom_id=self.custom_id_value,
            )
        )

    async def interaction_check(self, interaction: discord.Interaction) -> bool:
        return True

    async def on_timeout(self):
        return

async def find_existing_panel(channel: discord.TextChannel, custom_id: str):
    async for msg in channel.history(limit=20):
        if msg.author.bot and msg.components:
            for row in msg.components:
                for child in row.children:
                    if getattr(child, "custom_id", None) == custom_id:
                        return msg
    return None

async def upsert_panel(bot, channel_id: int, embed: discord.Embed, custom_id: str, label: str, emoji: str):
    if channel_id == 0:
        return False

    channel = bot.get_channel(channel_id)
    if channel is None:
        try:
            channel = await bot.fetch_channel(channel_id)
        except Exception:
            return False

    if not isinstance(channel, discord.TextChannel):
        return False

    view = discord.ui.View(timeout=None)
    view.add_item(discord.ui.Button(
        label=label,
        emoji=emoji,
        style=discord.ButtonStyle.secondary,
        custom_id=custom_id
    ))

    existing = await find_existing_panel(channel, custom_id)
    if existing:
        await existing.edit(embed=embed, view=view)
    else:
        await channel.send(embed=embed, view=view)
    return True

async def post_mobile_join(bot):

class DiffMobileUIPanelPack(commands.Cog):
    def __init__(self, bot):

# ===============================
# SETUP
# ===============================

def setup_diff_mobile_ui_panel_pack(bot: commands.Bot):
    ensure_data_file()

    @bot.event
    async def on_ready():
        if getattr(bot, "_diff_mobile_ui_ready_once", False):
            return
        bot._diff_mobile_ui_ready_once = True

        await bot.add_cog(DiffMobileUIPanelPack(bot))
        await post_mobile_join(bot)
        await post_mobile_host(bot)
        await post_mobile_hub(bot)
        await post_mobile_leaderboard(bot)
        await post_mobile_support(bot)

        print("[DIFF MOBILE UI] panel pack ready.")

    @bot.event
    async def on_interaction(interaction: discord.Interaction):
        if not interaction.type == discord.InteractionType.component:
            return

        cid = interaction.data.get("custom_id")
        if cid == JOIN_REFRESH_ID:
            await post_mobile_join(bot)
            await interaction.response.send_message("Join panel refreshed.", ephemeral=True)
        elif cid == HOST_REFRESH_ID:
            await post_mobile_host(bot)
            await interaction.response.send_message("Host panel refreshed.", ephemeral=True)
        elif cid == HUB_REFRESH_ID:
            await post_mobile_hub(bot)
            await interaction.response.send_message("Crew hub refreshed.", ephemeral=True)
        elif cid == LB_REFRESH_ID:
            await post_mobile_leaderboard(bot)
            await interaction.response.send_message("Leaderboard refreshed.", ephemeral=True)
        elif cid == SUPPORT_REFRESH_ID:
            await post_mobile_support(bot)
            await interaction.response.send_message("Support panel refreshed.", ephemeral=True)
