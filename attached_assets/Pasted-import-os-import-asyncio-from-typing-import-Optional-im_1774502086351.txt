import os
import asyncio
from typing import Optional

import discord
from discord.ext import commands
from discord import app_commands

# =========================================================
# DIFF Crew Announcement Panel
# ---------------------------------------------------------
# What this file does:
# - Auto-posts a single announcement panel in your target channel
# - Deletes old bot-made copies first (anti-dupe)
# - Lets staff press a button to open a modal
# - Sends the crew announcement and pings the crew role
#
# How to use:
# 1) Add this file to your bot project
# 2) Change the ROLE IDs below
# 3) Make sure your bot has:
#    - Send Messages
#    - Manage Messages
#    - Use Slash Commands
#    - Mention the crew role
# 4) Load the cog in your main bot file
# =========================================================

ANNOUNCEMENT_PANEL_CHANNEL_ID = 990097152044855326

# CHANGE THESE:
CREW_ROLE_ID = 123456789012345678      # Role to ping for crew announcements
STAFF_ROLE_ID = 123456789012345678     # Staff role allowed to use the button

PANEL_TITLE = "📢 DIFF Crew Announcement Center"
PANEL_DESCRIPTION = (
    "**Staff Announcement Panel**\n\n"
    "Use the button below to create and send an official crew announcement.\n\n"
    "This panel is for **staff use only**.\n"
    "When an announcement is submitted, the bot will post it here and ping the crew members automatically.\n\n"
    "━━━━━━━━━━━━━━━━━━━━━━\n"
    "• Keep announcements clear and important\n"
    "• Use short titles when possible\n"
    "• Avoid spam or duplicate posts\n"
    "━━━━━━━━━━━━━━━━━━━━━━"
)


class CrewAnnouncementModal(discord.ui.Modal, title="Create Crew Announcement"):
    title_input = discord.ui.TextInput(
        label="Announcement Title",
        placeholder="Example: Saturday Meet Update",
        max_length=100,
        required=True,
    )

    message_input = discord.ui.TextInput(
        label="Announcement Message",
        placeholder="Type the full crew announcement here...",
        style=discord.TextStyle.paragraph,
        max_length=2000,
        required=True,
    )

    footer_input = discord.ui.TextInput(
        label="Footer Text (optional)",
        placeholder="Example: DIFF Staff Team",
        max_length=100,
        required=False,
    )

    def __init__(self, cog: "CrewAnnouncementPanelCog"):
        super().__init__()
        self.cog = cog

    async def on_submit(self, interaction: discord.Interaction) -> None:
        member = interaction.user
        if not self.cog.user_is_staff(member):
            await interaction.response.send_message(
                "You do not have permission to use this panel.",
                ephemeral=True,
            )
            return

        channel = interaction.client.get_channel(ANNOUNCEMENT_PANEL_CHANNEL_ID)
        if channel is None:
            await interaction.response.send_message(
                "I could not find the announcement channel.",
                ephemeral=True,
            )
            return

        embed = discord.Embed(
            title=str(self.title_input),
            description=str(self.message_input),
            color=discord.Color.red(),
        )
        embed.set_author(
            name=f"Posted by {member.display_name}",
            icon_url=member.display_avatar.url if member.display_avatar else discord.Embed.Empty,
        )

        footer_text = str(self.footer_input).strip() or "DIFF Crew Announcement System"
        embed.set_footer(text=footer_text)

        allowed_mentions = discord.AllowedMentions(roles=True)

        await channel.send(
            content=f"<@&{CREW_ROLE_ID}>",
            embed=embed,
            allowed_mentions=allowed_mentions,
        )

        await interaction.response.send_message(
            "Crew announcement posted successfully.",
            ephemeral=True,
        )


class CrewAnnouncementPanelView(discord.ui.View):
    def __init__(self, cog: "CrewAnnouncementPanelCog"):
        super().__init__(timeout=None)
        self.cog = cog

    @discord.ui.button(
        label="Make Crew Announcement",
        style=discord.ButtonStyle.red,
        emoji="📣",
        custom_id="diff:crew_announcement:create",
    )
    async def make_announcement(
        self,
        interaction: discord.Interaction,
        button: discord.ui.Button,
    ) -> None:
        if not self.cog.user_is_staff(interaction.user):
            await interaction.response.send_message(
                "Only staff can use this button.",
                ephemeral=True,
            )
            return

        await interaction.response.send_modal(CrewAnnouncementModal(self.cog))


class CrewAnnouncementPanelCog(commands.Cog):
    def __init__(self, bot: commands.Bot):
        self.bot = bot
        self.panel_view = CrewAnnouncementPanelView(self)

    def user_is_staff(self, user: discord.abc.User) -> bool:
        if not isinstance(user, discord.Member):
            return False
        return any(role.id == STAFF_ROLE_ID for role in user.roles) or user.guild_permissions.administrator

    async def post_or_refresh_panel(self) -> None:
        await self.bot.wait_until_ready()
        await asyncio.sleep(2)

        channel = self.bot.get_channel(ANNOUNCEMENT_PANEL_CHANNEL_ID)
        if channel is None or not isinstance(channel, discord.TextChannel):
            print("Crew announcement panel channel not found.")
            return

        # Delete older bot-made panel messages so only one stays up.
        try:
            async for msg in channel.history(limit=50):
                if msg.author.id != self.bot.user.id:
                    continue

                has_panel_embed = False
                if msg.embeds:
                    for emb in msg.embeds:
                        if emb.title == PANEL_TITLE:
                            has_panel_embed = True
                            break

                has_panel_button = False
                for row in msg.components:
                    for component in row.children:
                        if getattr(component, "custom_id", None) == "diff:crew_announcement:create":
                            has_panel_button = True
                            break

                if has_panel_embed or has_panel_button:
                    await msg.delete()
        except discord.Forbidden:
            print("Missing permission to manage/delete old panel messages.")
        except Exception as e:
            print(f"Error cleaning old crew announcement panels: {e}")

        embed = discord.Embed(
            title=PANEL_TITLE,
            description=PANEL_DESCRIPTION,
            color=discord.Color.dark_red(),
        )
        embed.set_footer(text="DIFF Meets • Crew Announcement System")

        await channel.send(embed=embed, view=self.panel_view)

    @commands.Cog.listener()
    async def on_ready(self) -> None:
        # Re-register persistent view every startup
        self.bot.add_view(self.panel_view)

        # Refresh panel each time the bot starts
        try:
            await self.post_or_refresh_panel()
        except Exception as e:
            print(f"Failed to refresh crew announcement panel: {e}")

    @app_commands.command(name="refresh_crew_announcement_panel", description="Refresh the crew announcement panel.")
    async def refresh_crew_announcement_panel(self, interaction: discord.Interaction) -> None:
        if not self.user_is_staff(interaction.user):
            await interaction.response.send_message(
                "Only staff can refresh this panel.",
                ephemeral=True,
            )
            return

        await self.post_or_refresh_panel()
        await interaction.response.send_message(
            "Crew announcement panel refreshed.",
            ephemeral=True,
        )


async def setup(bot: commands.Bot):
    await bot.add_cog(CrewAnnouncementPanelCog(bot))
