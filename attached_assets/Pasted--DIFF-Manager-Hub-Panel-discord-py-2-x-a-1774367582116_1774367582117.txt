# =========================================================
# HELPERS
# =========================================================

def load_state() -> dict:
    if not STATE_FILE.exists():
        data = {"channel_id": MANAGER_HUB_CHANNEL_ID, "message_id": None}
        save_state(data)
        return data
    with STATE_FILE.open("r", encoding="utf-8") as f:
        return json.load(f)

def save_state(data: dict) -> None:
    with STATE_FILE.open("w", encoding="utf-8") as f:
        json.dump(data, f, indent=4)

class ManagerHubSelect(discord.ui.Select):
    def __init__(self):
        options = [
            discord.SelectOption(
                label=section["label"][:100],
                description="Open this manager hub section.",
                value=key,
                emoji=section["emoji"],
            )
            for key, section in SECTION_CONTENT.items()
        ]

        super().__init__(
            placeholder="Select a manager section to read...",
            min_values=1,
            max_values=1,
            options=options,
            custom_id="diff_manager_hub_select",
        )

    async def callback(self, interaction: discord.Interaction):
        key = self.values[0]
        section = SECTION_CONTENT.get(key)

        if not section:
            await interaction.response.send_message(
                "That section could not be found. Try refreshing the panel.",
                ephemeral=True,
            )
            return

        embed = discord.Embed(
            title=f"{section['emoji']} {section['label']}",
            description=section["description"],
            color=discord.Color.blurple(),
        )
        embed.set_footer(text=FOOTER_TEXT)

        if LOGO_URL:
            embed.set_thumbnail(url=LOGO_URL)
        if BANNER_URL:
            embed.set_image(url=BANNER_URL)

        await interaction.response.send_message(embed=embed, ephemeral=True)


class ManagerHubView(discord.ui.View):
    def __init__(self):
        super().__init__(timeout=None)
        self.add_item(ManagerHubSelect())


# =========================================================
# COG
# =========================================================

class ManagerHubSystem(commands.Cog):
    def __init__(self, bot: commands.Bot):
        self.bot = bot
        self.bot.add_view(ManagerHubView())

    def build_main_embed(self) -> discord.Embed:
        embed = discord.Embed(
            title=HUB_TITLE,
            description=HUB_DESCRIPTION,
            color=discord.Color.blurple(),
        )
        embed.add_field(
            name="Included Sections",
            value=(
                "• Crew Managers Roles & Responsibility\n"
                "• Crew Recruitment Roles\n"
                "• Discord Manager Roles\n"
                "• PlayStation Group Chat Manager Roles\n"
                "• Problem To Look Out For"
            ),
            inline=False,
        )
        embed.add_field(
            name="How To Use",
            value="Use the dropdown menu below to open each section in a clean embed.",
            inline=False,
        )
        embed.set_footer(text=FOOTER_TEXT)

        if LOGO_URL:
            embed.set_thumbnail(url=LOGO_URL)
        if BANNER_URL:
            embed.set_image(url=BANNER_URL)

        return embed

    async def post_or_refresh_panel(self):
        state = load_state()
        channel_id = state.get("channel_id", MANAGER_HUB_CHANNEL_ID)
        message_id = state.get("message_id")

        channel = self.bot.get_channel(channel_id)
        if channel is None:
            try:
                channel = await self.bot.fetch_channel(channel_id)
            except Exception as e:
                return False, f"Could not access the manager hub channel: {e}"

        embed = self.build_main_embed()
        view = ManagerHubView()
        content = f"<@&{MANAGER_ROLE_PING_ID}>" if MANAGER_ROLE_PING_ID else None

        if message_id:
            try:
                msg = await channel.fetch_message(message_id)
                await msg.edit(content=content, embed=embed, view=view)
                return True, "Manager hub panel refreshed successfully."
            except Exception:
                pass

        try:
            msg = await channel.send(content=content, embed=embed, view=view)
            state["message_id"] = msg.id
            state["channel_id"] = channel.id
            save_state(state)
            return True, "Manager hub panel posted successfully."
        except Exception as e:
            return False, f"Failed to post manager hub panel: {e}"


# =========================================================
# SLASH COMMANDS
# =========================================================

class ManagerHubGroup(app_commands.Group):
    def __init__(self, cog: ManagerHubSystem):
        super().__init__(name="managerhub", description="Manage the DIFF Manager Hub panel")
        self.cog = cog

    async def interaction_check(self, interaction: discord.Interaction) -> bool:
        if not interaction.user.guild_permissions.manage_guild:
            await interaction.response.send_message(
                "You need Manage Server to use this.",
                ephemeral=True,
            )
            return False
        return True

    @app_commands.command(name="post", description="Post or refresh the manager hub panel")
    async def post_cmd(self, interaction: discord.Interaction):
        ok, msg = await self.cog.post_or_refresh_panel()
        await interaction.response.send_message(msg, ephemeral=True)

    @app_commands.command(name="refresh", description="Refresh the manager hub panel")
    async def refresh_cmd(self, interaction: discord.Interaction):
        ok, msg = await self.cog.post_or_refresh_panel()
        await interaction.response.send_message(msg, ephemeral=True)


# =========================================================
# SETUP
# =========================================================

async def setup(bot: commands.Bot):
    cog = ManagerHubSystem(bot)
    await bot.add_cog(cog, guild=discord.Object(id=GUILD_ID))
    bot.tree.add_command(ManagerHubGroup(cog), guild=discord.Object(id=GUILD_ID))
