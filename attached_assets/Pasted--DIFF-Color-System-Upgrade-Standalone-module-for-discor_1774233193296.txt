# ============================================================
# STATE
# ============================================================
@dataclass
class ColorVoteState:
    title: str
    color_name: str
    image_url: str
    created_by_id: int
    channel_id: int
    message_id: Optional[int] = None
    end_time: Optional[datetime] = None
    is_closed: bool = False
    votes: Dict[int, str] = field(default_factory=dict)  # user_id -> yes/no

    @property
    def yes_count(self) -> int:
        return sum(1 for v in self.votes.values() if v == "yes")

    @property
    def no_count(self) -> int:
        return sum(1 for v in self.votes.values() if v == "no")

    @property
    def total_votes(self) -> int:
        return len(self.votes)


ACTIVE_VOTES: Dict[int, ColorVoteState] = {}
ANNOUNCE_LOCK = asyncio.Lock()


# ============================================================
# HELPERS
# ============================================================
def utc_now() -> datetime:
    return datetime.now(timezone.utc)


def has_staff_access(member: discord.Member) -> bool:
    allowed_roles = {LEADER_ROLE_ID, CO_LEADER_ROLE_ID, MANAGER_ROLE_ID}
    return any(role.id in allowed_roles for role in member.roles)


def format_dt(dt: datetime) -> str:
    ts = int(dt.timestamp())
    return f"<t:{ts}:F> (<t:{ts}:R>)"


def build_vote_embed(state: ColorVoteState) -> discord.Embed:
    embed = discord.Embed(
        title="🎨 DIFF Crew Color Voting",
        description=(
            f"Vote on this week's crew color selection.\n\n"
            f"**Color Name:** *{state.color_name}*\n"
            f"**Status:** {'Closed' if state.is_closed else 'Open'}"
        ),
        color=ACCENT_COLOR,
        timestamp=utc_now(),
    )
    if state.image_url:
        embed.set_image(url=state.image_url)
    embed.add_field(name="✅ Approve", value=str(state.yes_count), inline=True)
    embed.add_field(name="❌ Pass", value=str(state.no_count), inline=True)
    embed.add_field(name="📊 Total Votes", value=str(state.total_votes), inline=True)
    if state.end_time and not state.is_closed:
        embed.add_field(name="⏰ Voting Ends", value=format_dt(state.end_time), inline=False)
    embed.set_thumbnail(url=DIFF_LOGO_URL)
    embed.set_footer(text=EMBED_FOOTER)
    return embed

async def safe_edit_vote_message(bot: commands.Bot, state: ColorVoteState) -> None:
    if not state.message_id:
        return
    channel = bot.get_channel(state.channel_id)
    if not isinstance(channel, discord.TextChannel):
        return
    try:
        msg = await channel.fetch_message(state.message_id)
        await msg.edit(embed=build_vote_embed(state), view=ColorVoteView(bot, state.message_id))
    except discord.NotFound:
        pass
    except discord.HTTPException:
        pass


async def close_vote_and_announce(bot: commands.Bot, message_id: int, closed_by: Optional[int] = None) -> bool:
    async with ANNOUNCE_LOCK:
        state = ACTIVE_VOTES.get(message_id)
        if not state or state.is_closed:
            return False

        state.is_closed = True
        await safe_edit_vote_message(bot, state)

        channel = bot.get_channel(state.channel_id)
        if not isinstance(channel, discord.TextChannel):
            return False
