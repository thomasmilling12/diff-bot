"""
DIFF Unverified Announcement Panel
Replit-ready discord.py extension/module

What this file does:
- Auto-posts an unverified announcement panel in the target channel
- Refreshes the same panel without creating duplicates
- Adds a staff-only button to ping the unverified role
- Re-registers the button on bot restart
- Stores the panel message ID in a small JSON file

Setup:
1) Put this file in your Replit project
2) Change UNVERIFIED_ROLE_ID to your actual role ID
3) In your main bot file, load this extension:
       await bot.load_extension("unverified_panel")
   or if using older sync setup:
       bot.load_extension("unverified_panel")

Requirements:
- discord.py 2.x
- Message Content intent is not required for this file
"""

import json
import os
from typing import Optional

import discord
from discord.ext import commands

# =========================
# CONFIG
# =========================
TARGET_CHANNEL_ID = 1486058055991824655
UNVERIFIED_ROLE_ID = 0  # <-- PUT YOUR UNVERIFIED ROLE ID HERE
DATA_FILE = "unverified_panel_data.json"

# Optional styling
EMBED_COLOR = 0x2B2D31
FOOTER_TEXT = "DIFF Verification System"
THUMBNAIL_URL = ""  # Optional: paste your logo URL here


# =========================
# STORAGE HELPERS
# =========================
def load_data() -> dict:
    if not os.path.exists(DATA_FILE):
        return {}
    try:
        with open(DATA_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    except Exception:
        return {}


def save_data(data: dict) -> None:
    with open(DATA_FILE, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=4)


def get_saved_message_id() -> Optional[int]:
    data = load_data()
    msg_id = data.get("panel_message_id")
    return int(msg_id) if msg_id else None


def save_message_id(message_id: int) -> None:
    data = load_data()
    data["panel_message_id"] = message_id
    save_data(data)


# =========================
# VIEW / BUTTON
# =========================
class UnverifiedPingView(discord.ui.View):
    def __init__(self, cog: "UnverifiedPanelCog"):
        super().__init__(timeout=None)
        self.cog = cog

    @discord.ui.button(
        label="Ping Unverified Role",
        style=discord.ButtonStyle.danger,
        emoji="📣",
        custom_id="diff_unverified_ping_button_v1",
    )
    async def ping_unverified(
        self, interaction: discord.Interaction, button: discord.ui.Button
    ):
        if not interaction.guild:
            return await interaction.response.send_message(
                "This button can only be used in a server.",
                ephemeral=True,
            )

        member = interaction.user
        if not isinstance(member, discord.Member):
            return await interaction.response.send_message(
                "Could not verify your staff permissions.",
                ephemeral=True,
            )

        # Staff-only permission check
        if not (
            member.guild_permissions.manage_guild
            or member.guild_permissions.manage_messages
            or member.guild_permissions.administrator
        ):
            return await interaction.response.send_message(
                "You need staff permissions to use this button.",
                ephemeral=True,
            )

        if UNVERIFIED_ROLE_ID == 0:
            return await interaction.response.send_message(
                "UNVERIFIED_ROLE_ID is still set to 0. Add your role ID in the file first.",
                ephemeral=True,
            )

        role = interaction.guild.get_role(UNVERIFIED_ROLE_ID)
        if role is None:
            return await interaction.response.send_message(
                "I could not find the unverified role. Double-check the role ID.",
                ephemeral=True,
            )

        await interaction.response.defer(ephemeral=True)

        channel = interaction.channel
        if not isinstance(channel, discord.abc.Messageable):
            return await interaction.followup.send(
                "I couldn't send the role ping in this channel.",
                ephemeral=True,
            )

        ping_embed = discord.Embed(
            title="📌 Verification Reminder",
            description=(
                f"{role.mention}\n\n"
                "Please complete the server verification process to unlock access.\n"
                "Read the required channels and follow the verification steps posted by staff."
            ),
            color=discord.Color.red(),
        )
        ping_embed.set_footer(text=f"Sent by {member.display_name}")

        await channel.send(
            content=role.mention,
            embed=ping_embed,
            allowed_mentions=discord.AllowedMentions(roles=True),
        )

        await interaction.followup.send(
            "Unverified role ping sent.",
            ephemeral=True,
        )


# =========================
# COG
# =========================
class UnverifiedPanelCog(commands.Cog):
    def __init__(self, bot: commands.Bot):
        self.bot = bot
        self.view = UnverifiedPingView(self)
        self.bot.add_view(self.view)

    def build_embed(self) -> discord.Embed:
        embed = discord.Embed(
            title="📍 DIFF Unverified Announcement Panel",
            description=(
                "Welcome to the **unverified** area.\n\n"
                "This panel is used by staff to send a quick reminder to all members "
                "who still need to complete verification.\n\n"
                "**What happens when staff press the button:**\n"
                "• The unverified role gets pinged\n"
                "• A reminder message is posted in this channel\n"
                "• Members are told to complete verification steps\n\n"
                "**For members:**\n"
                "Please follow the server verification process so you can gain full access."
            ),
            color=EMBED_COLOR,
        )

        embed.add_field(
            name="Staff Use",
            value=(
                "Press the button below when you need to remind unverified members "
                "to finish verification."
            ),
            inline=False,
        )

        embed.add_field(
            name="Anti-Dupe System",
            value=(
                "This panel is tracked by message ID and will refresh the same post "
                "instead of making duplicates."
            ),
            inline=False,
        )

        if THUMBNAIL_URL:
            embed.set_thumbnail(url=THUMBNAIL_URL)

        embed.set_footer(text=FOOTER_TEXT)
        return embed

    async def get_target_channel(self) -> Optional[discord.TextChannel]:
        channel = self.bot.get_channel(TARGET_CHANNEL_ID)
        if channel is None:
            try:
                fetched = await self.bot.fetch_channel(TARGET_CHANNEL_ID)
                if isinstance(fetched, discord.TextChannel):
                    return fetched
                return None
            except Exception:
                return None
        return channel if isinstance(channel, discord.TextChannel) else None

    async def ensure_panel(self) -> None:
        channel = await self.get_target_channel()
        if channel is None:
            print(f"[UnverifiedPanel] Could not find channel: {TARGET_CHANNEL_ID}")
            return

        embed = self.build_embed()
        saved_message_id = get_saved_message_id()

        # Try editing existing tracked message first
        if saved_message_id:
            try:
                msg = await channel.fetch_message(saved_message_id)
                await msg.edit(embed=embed, view=self.view)
                print("[UnverifiedPanel] Existing panel refreshed.")
                return
            except discord.NotFound:
                pass
            except Exception as e:
                print(f"[UnverifiedPanel] Failed to edit saved panel: {e}")

        # Cleanup any old bot-created panel duplicates in the channel
        try:
            async for msg in channel.history(limit=30):
                if (
                    msg.author == self.bot.user
                    and msg.embeds
                    and msg.embeds[0].title == "📍 DIFF Unverified Announcement Panel"
                ):
                    try:
                        await msg.delete()
                    except Exception:
                        pass
        except Exception as e:
            print(f"[UnverifiedPanel] Duplicate cleanup skipped: {e}")

        # Post a fresh panel
        try:
            new_msg = await channel.send(embed=embed, view=self.view)
            save_message_id(new_msg.id)
            print(f"[UnverifiedPanel] New panel posted: {new_msg.id}")
        except Exception as e:
            print(f"[UnverifiedPanel] Failed to post panel: {e}")

    @commands.Cog.listener()
    async def on_ready(self):
        # Prevent noisy repeated logs on reconnects, but still refresh safely
        await self.ensure_panel()

    @commands.command(name="refresh_unverified_panel")
    @commands.has_permissions(manage_guild=True)
    async def refresh_unverified_panel(self, ctx: commands.Context):
        await self.ensure_panel()
        await ctx.send("Unverified panel refreshed.", delete_after=10)


# =========================
# EXTENSION SETUP
# =========================
async def setup(bot: commands.Bot):
    await bot.add_cog(UnverifiedPanelCog(bot))
