"""
DIFF Host Control Hub
Standalone discord.py module for posting and updating the DIFF Host Hub panel.

What this file adds:
- Main "DIFF Host Control Hub" panel
- 5 buttons:
    1) Host Guide
    2) Role Assignments
    3) Meet Reminders
    4) Host Checklist
    5) Blacklist Check
- Update-in-place support so reposting edits the existing panel instead of duplicating it
- Persistent button view support for discord.py 2.x bots

Setup:
1) Put this file in your bot project.
2) Update HOST_HUB_CHANNEL_ID below.
3) Update BLACKLIST_CHANNEL_URL below.
4) Load the cog:
      await bot.add_cog(HostHubCog(bot))
5) In on_ready, register the persistent view:
      bot.add_view(HostHubView(bot))
6) Use:
      !post_host_hub
   or:
      /post_host_hub

Requirements:
- discord.py 2.x
- Send Messages, Embed Links, Use Application Commands permissions
"""

HOST_HUB_CHANNEL_ID = 1134690348220825730  # <-- replace if needed
BLACKLIST_CHANNEL_URL = "https://discord.com/channels/850386896509337710/1057016810261712938"

# JSON file used to remember the panel message ID so the bot edits instead of reposting
STATE_FILE = Path("host_hub_state.json")


# =========================
# STATE HELPERS
# =========================

def load_state() -> dict:
    if not STATE_FILE.exists():
        return {}
    try:
        return json.loads(STATE_FILE.read_text(encoding="utf-8"))
    except Exception:
        return {}


def save_state(data: dict) -> None:
    STATE_FILE.write_text(json.dumps(data, indent=2), encoding="utf-8")


def get_saved_message_id() -> Optional[int]:
    data = load_state()
    value = data.get("host_hub_message_id")
    return int(value) if value else None


def set_saved_message_id(message_id: int) -> None:
    data = load_state()
    data["host_hub_message_id"] = message_id
    save_state(data)

class HostHubCog(commands.Cog):
    def __init__(self, bot: commands.Bot):
        self.bot = bot

    async def post_or_update_panel(self, destination: discord.abc.Messageable) -> discord.Message:
        embed = main_panel_embed()
        view = HostHubView(self.bot)

        saved_message_id = get_saved_message_id()

        # Try editing the saved message first
        if saved_message_id and isinstance(destination, discord.TextChannel):
            try:
                old_message = await destination.fetch_message(saved_message_id)
                await old_message.edit(embed=embed, view=view)
                return old_message
            except Exception:
                pass

        # Otherwise create a new one and save it
        new_message = await destination.send(embed=embed, view=view)
        set_saved_message_id(new_message.id)
        return new_message

    @commands.command(name="post_host_hub")
    @commands.has_permissions(manage_guild=True)
    async def post_host_hub_prefix(self, ctx: commands.Context):
        """Post or update the host hub panel in the configured host hub channel."""
        channel = self.bot.get_channel(HOST_HUB_CHANNEL_ID)
        if channel is None:
            await ctx.reply("I couldn't find the configured host hub channel. Please check HOST_HUB_CHANNEL_ID.")
            return

        message = await self.post_or_update_panel(channel)
        await ctx.reply(f"Host hub posted/updated: {message.jump_url}", mention_author=False)

    @app_commands.command(name="post_host_hub", description="Post or update the DIFF Host Control Hub panel.")
    async def post_host_hub_slash(self, interaction: discord.Interaction):
        if not interaction.user.guild_permissions.manage_guild:
            await interaction.response.send_message("You do not have permission to use this command.", ephemeral=True)
            return

        channel = self.bot.get_channel(HOST_HUB_CHANNEL_ID)
        if channel is None:
            await interaction.response.send_message(
                "I couldn't find the configured host hub channel. Please check HOST_HUB_CHANNEL_ID.",
                ephemeral=True,
            )
            return

        message = await self.post_or_update_panel(channel)
        await interaction.response.send_message(
            f"Host hub posted/updated successfully: {message.jump_url}",
            ephemeral=True,
        )


async def setup(bot: commands.Bot):
    await bot.add_cog(HostHubCog(bot))
    bot.add_view(HostHubView(bot))
