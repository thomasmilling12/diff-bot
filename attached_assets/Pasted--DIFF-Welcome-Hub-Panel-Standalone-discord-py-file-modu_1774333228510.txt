DIFF Welcome Hub Panel
Standalone discord.py file/module for posting and refreshing a welcome hub panel.

How to use:
1) Put this file into your bot project.
2) Make sure you have discord.py 2.x installed.
3) Replace the placeholder channel IDs below if needed.
4) Import and call `setup_welcome_hub(bot)` in your main bot file.
5) Run the text command: !welcomehub
   - It will post or refresh the panel in the configured channel.
   - It will reuse the same message instead of creating duplicates.

Target post channel:
https://discord.com/channels/850386896509337710/1485687906382123331
"""

# =========================

# =========================
def channel_mention(channel_id: Optional[int], fallback_text: str) -> str:
    if channel_id:
        return f"<#{channel_id}>"
    return fallback_text


def load_message_id() -> Optional[int]:
    if not STATE_FILE.exists():
        return None
    try:
        data = json.loads(STATE_FILE.read_text(encoding="utf-8"))
        message_id = data.get("welcome_hub_message_id")
        return int(message_id) if message_id else None
    except Exception:
        return None


def save_message_id(message_id: int) -> None:
    STATE_FILE.write_text(
        json.dumps({"welcome_hub_message_id": message_id}, indent=2),
        encoding="utf-8",
    )

# =========================
# POST / REFRESH LOGIC
# =========================
async def post_or_refresh_welcome_hub(bot: commands.Bot) -> discord.Message:
    channel = bot.get_channel(WELCOME_HUB_CHANNEL_ID)
    if channel is None:
        channel = await bot.fetch_channel(WELCOME_HUB_CHANNEL_ID)

    if not isinstance(channel, discord.TextChannel):
        raise RuntimeError("Configured welcome hub channel is not a text channel.")

    embed = build_welcome_embed()
    view = WelcomeHubView()

    existing_message_id = load_message_id()
    if existing_message_id:
        try:
            existing_message = await channel.fetch_message(existing_message_id)
            await existing_message.edit(embed=embed, view=view)
            return existing_message
        except discord.NotFound:
            pass
        except discord.HTTPException:
            pass

    new_message = await channel.send(embed=embed, view=view)
    save_message_id(new_message.id)
    return new_message


# =========================
# COG
# =========================
class WelcomeHub(commands.Cog):
    def __init__(self, bot: commands.Bot) -> None:
        self.bot = bot
        self.bot.add_view(WelcomeHubView())  # persistent buttons

    @commands.command(name="welcomehub")
    @commands.has_permissions(administrator=True)
    async def welcomehub_command(self, ctx: commands.Context) -> None:
        try:
            message = await post_or_refresh_welcome_hub(self.bot)
            await ctx.reply(
                f"Welcome Hub has been posted/refreshed in <#{WELCOME_HUB_CHANNEL_ID}>.\n"
                f"Message ID: `{message.id}`",
                mention_author=False,
            )
        except Exception as exc:
            await ctx.reply(f"Failed to post Welcome Hub: `{exc}`", mention_author=False)

    @commands.Cog.listener()
    async def on_ready(self) -> None:
        # Ensures persistent view is registered after restart
        self.bot.add_view(WelcomeHubView())


async def setup(bot: commands.Bot) -> None:
    await bot.add_cog(WelcomeHub(bot))
