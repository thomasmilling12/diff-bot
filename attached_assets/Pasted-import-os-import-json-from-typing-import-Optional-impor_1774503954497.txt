import os
import json
from typing import Optional

import discord
from discord.ext import commands, tasks
from dotenv import load_dotenv

load_dotenv()

# =========================
# DIFF Server Stats Panel
# =========================
# Drop this into your current bot project.
# Works with discord.py 2.x
#
# What it does:
# - Posts ONE server stats panel in your chosen channel
# - Auto-refreshes every 5 minutes
# - Recreates the panel if it gets deleted
# - Prevents duplicates by saving the message ID
#
# Setup:
# 1) Make sure your bot has these intents enabled:
#    - members
#    - presences   (optional, but needed for online count)
# 2) Make sure the bot can:
#    - View Channel
#    - Send Messages
#    - Embed Links
#    - Read Message History
# 3) Add this file to your bot
# 4) Load the cog in your main bot file
#
# Example for main.py:
#   await bot.load_extension("server_stats_panel")
#
# Or if using sync load:
#   bot.load_extension("server_stats_panel")
#
# Channel used:
# https://discord.com/channels/850386896509337710/1485273802391814224
# Channel ID = 1485273802391814224

TARGET_CHANNEL_ID = 1485273802391814224
STATS_DATA_FILE = "server_stats_panel_data.json"


def load_stats_data() -> dict:
    if not os.path.exists(STATS_DATA_FILE):
        return {}
    try:
        with open(STATS_DATA_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    except Exception:
        return {}


def save_stats_data(data: dict) -> None:
    with open(STATS_DATA_FILE, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=2)


class ServerStatsPanel(commands.Cog):
    def __init__(self, bot: commands.Bot):
        self.bot = bot
        self.stats_updater.start()

    def cog_unload(self):
        self.stats_updater.cancel()

    def get_saved_message_id(self, guild_id: int) -> Optional[int]:
        data = load_stats_data()
        guild_data = data.get(str(guild_id), {})
        return guild_data.get("message_id")

    def set_saved_message_id(self, guild_id: int, message_id: int) -> None:
        data = load_stats_data()
        if str(guild_id) not in data:
            data[str(guild_id)] = {}
        data[str(guild_id)]["message_id"] = message_id
        save_stats_data(data)

    def clear_saved_message_id(self, guild_id: int) -> None:
        data = load_stats_data()
        if str(guild_id) in data and "message_id" in data[str(guild_id)]:
            del data[str(guild_id)]["message_id"]
            save_stats_data(data)

    def build_embed(self, guild: discord.Guild) -> discord.Embed:
        members = guild.members

        total_members = guild.member_count or len(members)
        bot_count = sum(1 for m in members if m.bot)
        human_count = total_members - bot_count

        online_count = sum(
            1 for m in members
            if getattr(m, "status", discord.Status.offline) != discord.Status.offline
        )

        boost_count = guild.premium_subscription_count or 0
        channel_count = len(guild.channels)
        role_count = len(guild.roles) - 1 if len(guild.roles) > 0 else 0

        embed = discord.Embed(
            title="📊 DIFF Server Statistics",
            description="Live server stats panel for the community.",
            color=discord.Color.blurple()
        )

        embed.add_field(name="👥 Total Members", value=f"**{total_members:,}**", inline=True)
        embed.add_field(name="🙋 Humans", value=f"**{human_count:,}**", inline=True)
        embed.add_field(name="🤖 Bots", value=f"**{bot_count:,}**", inline=True)

        embed.add_field(name="🟢 Online Now", value=f"**{online_count:,}**", inline=True)
        embed.add_field(name="💎 Boosts", value=f"**{boost_count:,}**", inline=True)
        embed.add_field(name="🎭 Roles", value=f"**{role_count:,}**", inline=True)

        embed.add_field(name="📁 Channels", value=f"**{channel_count:,}**", inline=True)
        embed.add_field(name="🆔 Server ID", value=f"`{guild.id}`", inline=True)
        embed.add_field(name="👑 Owner", value=f"{guild.owner.mention if guild.owner else 'Unknown'}", inline=True)

        if guild.icon:
            embed.set_thumbnail(url=guild.icon.url)

        embed.set_footer(text="Auto-refreshes every 5 minutes • No duplicate panels")
        embed.timestamp = discord.utils.utcnow()
        return embed

    async def ensure_panel(self, guild: discord.Guild) -> None:
        channel = guild.get_channel(TARGET_CHANNEL_ID)
        if channel is None:
            return
        if not isinstance(channel, discord.TextChannel):
            return

        embed = self.build_embed(guild)
        saved_message_id = self.get_saved_message_id(guild.id)

        if saved_message_id:
            try:
                msg = await channel.fetch_message(saved_message_id)
                await msg.edit(embed=embed, content=None)
                return
            except discord.NotFound:
                self.clear_saved_message_id(guild.id)
            except discord.Forbidden:
                return
            except discord.HTTPException:
                pass

        # Recreate the panel if missing
        try:
            new_msg = await channel.send(embed=embed)
            self.set_saved_message_id(guild.id, new_msg.id)
        except discord.Forbidden:
            pass
        except discord.HTTPException:
            pass

    @tasks.loop(minutes=5)
    async def stats_updater(self):
        for guild in self.bot.guilds:
            await self.ensure_panel(guild)

    @stats_updater.before_loop
    async def before_stats_updater(self):
        await self.bot.wait_until_ready()

    @commands.Cog.listener()
    async def on_ready(self):
        for guild in self.bot.guilds:
            await self.ensure_panel(guild)
        print("✅ Server Stats Panel loaded.")

    @commands.command(name="setupserverstats")
    @commands.has_permissions(administrator=True)
    async def setup_server_stats(self, ctx: commands.Context):
        """Manually create or refresh the stats panel."""
        if ctx.guild is None:
            return

        await self.ensure_panel(ctx.guild)
        await ctx.send("✅ Server stats panel has been posted/refreshed.", delete_after=10)

    @commands.command(name="refreshserverstats")
    @commands.has_permissions(administrator=True)
    async def refresh_server_stats(self, ctx: commands.Context):
        """Manually refresh the stats panel."""
        if ctx.guild is None:
            return

        await self.ensure_panel(ctx.guild)
        await ctx.send("🔄 Server stats panel refreshed.", delete_after=10)

    @setup_server_stats.error
    @refresh_server_stats.error
    async def stats_command_error(self, ctx: commands.Context, error):
        if isinstance(error, commands.MissingPermissions):
            await ctx.send("❌ You need Administrator permission to use this command.", delete_after=10)


async def setup(bot: commands.Bot):
    await bot.add_cog(ServerStatsPanel(bot))
