
import discord
from discord import app_commands
from discord.ext import commands


# ============================================================
# DIFF Color Submission Panel
# Panel Channel: 1177436572304556084
# Submission Channel: 1177434999381831680
# Works with discord.py 2.x
# ============================================================

COLOR_PANEL_CHANNEL_ID = 1177436572304556084
COLOR_SUBMISSION_CHANNEL_ID = 1177434999381831680


class ColorSubmissionModal(discord.ui.Modal, title="DIFF Color Submission"):
    color_name = discord.ui.TextInput(
        label="Color Name",
        placeholder="Example: Tangerine Tango",
        max_length=100,
        required=True,
    )

    hex_code = discord.ui.TextInput(
        label="HEX Code",
        placeholder="Example: #FF9742",
        max_length=7,
        min_length=4,
        required=True,
    )

    image_url = discord.ui.TextInput(
        label="Image URL",
        placeholder="Paste the direct image link here",
        style=discord.TextStyle.paragraph,
        required=True,
    )

    def __init__(self, bot: commands.Bot):
        super().__init__()
        self.bot = bot

    async def on_submit(self, interaction: discord.Interaction) -> None:
        submit_channel = self.bot.get_channel(COLOR_SUBMISSION_CHANNEL_ID)
        if submit_channel is None:
            submit_channel = await self.bot.fetch_channel(COLOR_SUBMISSION_CHANNEL_ID)

        color_name = str(self.color_name.value).strip()
        hex_code = str(self.hex_code.value).strip().upper()
        image_url = str(self.image_url.value).strip()

        if not hex_code.startswith("#"):
            hex_code = f"#{hex_code}"

        if len(hex_code) not in (4, 7):
            await interaction.response.send_message(
                "Your HEX code needs to look like `#FF9742` or `#F94`.",
                ephemeral=True,
            )
            return

        try:
            embed_color = discord.Color.from_str(hex_code)
        except Exception:
            embed_color = discord.Color.blurple()

        embed = discord.Embed(
            title="🎨 DIFF Color Submission",
            description=(
                "A new crew color has been submitted by the Color Team.\n\n"
                f"**Color Name:** {color_name}\n"
                f"**HEX Code:** `{hex_code}`\n\n"
                "Use this post to review the submission."
            ),
            color=embed_color,
        )
        embed.set_image(url=image_url)
        embed.set_footer(text=f"Submitted by {interaction.user.display_name}")

        msg = await submit_channel.send(embed=embed)
        for emoji in ("✅", "❌", "🤔"):
            try:
                await msg.add_reaction(emoji)
            except Exception:
                pass

        await interaction.response.send_message(
            f"Your color submission has been posted in {submit_channel.mention}.",
            ephemeral=True,
        )


class ColorSubmissionPanelView(discord.ui.View):
    def __init__(self, bot: commands.Bot):
        super().__init__(timeout=None)
        self.bot = bot

    @discord.ui.button(
        label="Submit Color",
        style=discord.ButtonStyle.primary,
        emoji="🎨",
        custom_id="diff_submit_color_button",
    )
    async def submit_color_button(
        self,
        interaction: discord.Interaction,
        button: discord.ui.Button,
    ) -> None:
        color_team_role = discord.utils.get(interaction.guild.roles, name="Color Team")

        # Optional role check:
        # If your role has a different name, change "Color Team" to match it.
        if color_team_role and color_team_role not in interaction.user.roles:
            await interaction.response.send_message(
                "Only the Color Team can submit crew colors.",
                ephemeral=True,
            )
            return

        await interaction.response.send_modal(ColorSubmissionModal(self.bot))


class ColorSubmissionPanelCog(commands.Cog):
    def __init__(self, bot: commands.Bot):
        self.bot = bot
        self.bot.add_view(ColorSubmissionPanelView(bot))

    @app_commands.command(
        name="post_color_submission_panel",
        description="Post the DIFF color submission panel.",
    )
    @app_commands.default_permissions(manage_guild=True)
    async def post_color_submission_panel(self, interaction: discord.Interaction) -> None:
        panel_channel = self.bot.get_channel(COLOR_PANEL_CHANNEL_ID)
        if panel_channel is None:
            panel_channel = await self.bot.fetch_channel(COLOR_PANEL_CHANNEL_ID)

        embed = discord.Embed(
            title="🎨 DIFF Color Submission Panel",
            description=(
                "**Color Team Guide**\n\n"
                "Use the button below to submit a new crew color for review.\n\n"
                "**What to include:**\n"
                "• Color name\n"
                "• HEX code\n"
                "• Image link for the preview car\n\n"
                "**Before submitting:**\n"
                "• Keep the color clean and realistic\n"
                "• Double-check the HEX code\n"
                "• Use a clear image that shows the color well\n"
                "• Make sure the submission is meet-ready\n\n"
                "**How it works:**\n"
                "• Press **Submit Color**\n"
                "• Fill out the form\n"
                "• Your submission will be posted in the review channel automatically\n"
                "• Review reactions will be added right away\n\n"
                "Press **Submit Color** below to begin."
            ),
            color=discord.Color.blurple(),
        )
        embed.set_footer(text="DIFF • Color Team System")

        await panel_channel.send(
            embed=embed,
            view=ColorSubmissionPanelView(self.bot),
        )

        await interaction.response.send_message(
            f"Color submission panel posted in {panel_channel.mention}.",
            ephemeral=True,
        )


async def setup(bot: commands.Bot):
    await bot.add_cog(ColorSubmissionPanelCog(bot))
