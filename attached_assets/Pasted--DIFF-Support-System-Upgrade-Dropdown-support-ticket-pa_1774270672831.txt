def clean_name(value: str) -> str:
    value = value.lower().strip()
    value = re.sub(r"[^a-z0-9]+", "-", value)
    value = re.sub(r"-{2,}", "-", value).strip("-")
    return value or "user"


def role_mention(role_id: Optional[int]) -> str:
    return f"<@&{role_id}>" if role_id else ""


def has_any_role(member: discord.Member, role_ids: set[int]) -> bool:
    return any(role.id in role_ids for role in member.roles)


def utc_now() -> datetime:
    return datetime.now(timezone.utc)


def format_dt(dt: datetime) -> str:
    unix = int(dt.timestamp())
    return f"<t:{unix}:F>"

def build_decision_embed(applicant: discord.Member, reviewer: discord.Member, approved: bool) -> discord.Embed:
    color = discord.Color.green() if approved else discord.Color.red()
    title = "✅ Application Approved" if approved else "❌ Application Denied"
    desc = (
        f"Applicant: {applicant.mention}\n"
        f"Reviewed By: {reviewer.mention}\n\n"
        f"Decision: {'Approved' if approved else 'Denied'}"
    )
    embed = discord.Embed(title=title, description=desc, color=color, timestamp=utc_now())
    return apply_branding(embed)


async def find_existing_ticket(guild: discord.Guild, member: discord.Member, ticket_type: TicketType) -> Optional[discord.TextChannel]:
    prefix = f"{ticket_type.channel_prefix}-{clean_name(member.name)}"
    for channel in guild.text_channels:
        if channel.name.startswith(prefix) and channel.topic:
            if f"ticket_owner={member.id}" in channel.topic and f"ticket_type={ticket_type.key}" in channel.topic:
                return channel
    return None


def parse_topic_value(topic: Optional[str], key: str) -> Optional[str]:
    if not topic:
        return None
    match = re.search(rf"{re.escape(key)}=([^|]+)", topic)
    if match:
        return match.group(1).strip()
    return None


async def export_transcript(channel: discord.TextChannel) -> discord.File:
    messages = []
    async for msg in channel.history(limit=None, oldest_first=True):
        created = msg.created_at.strftime("%Y-%m-%d %H:%M:%S UTC")
        header = f"[{created}] {msg.author} ({msg.author.id})"
        content = msg.content or ""
        attachments = ""
        if msg.attachments:
            attachments = "\n" + "\n".join(f"[Attachment] {a.url}" for a in msg.attachments)
        embeds = ""
        if msg.embeds:
            embed_lines = []
            for emb in msg.embeds:
                if emb.title:
                    embed_lines.append(f"[Embed Title] {emb.title}")
                if emb.description:
                    embed_lines.append(f"[Embed Description] {emb.description}")
                for field in emb.fields:
                    embed_lines.append(f"[Embed Field] {field.name}: {field.value}")
            if embed_lines:
                embeds = "\n" + "\n".join(embed_lines)
        messages.append(f"{header}\n{content}{attachments}{embeds}\n{'-'*60}\n")

    text = "".join(messages) if messages else "No messages found."
    buffer = io.BytesIO(text.encode("utf-8"))
    safe_name = channel.name[:80]
    return discord.File(buffer, filename=f"{safe_name}-transcript.txt")

class CloseTicketView(discord.ui.View):
    def __init__(self) -> None:
        super().__init__(timeout=None)

    @discord.ui.button(label="Close Ticket", emoji="🔒", style=discord.ButtonStyle.danger, custom_id="diff_support_close_ticket")
    async def close_ticket(self, interaction: discord.Interaction, _: discord.ui.Button) -> None:
        if not interaction.guild or not isinstance(interaction.channel, discord.TextChannel):
            await interaction.response.send_message("This can only be used in a ticket channel.", ephemeral=True)
            return
        if not isinstance(interaction.user, discord.Member):
            await interaction.response.send_message("Member check failed.", ephemeral=True)
            return

        channel = interaction.channel
        owner_id = parse_topic_value(channel.topic, "ticket_owner")
        is_owner = owner_id == str(interaction.user.id)
        staff_role_ids = {LEADER_ROLE_ID, CO_LEADER_ROLE_ID, MANAGER_ROLE_ID, HOST_ROLE_ID}
        if SUPPORT_STAFF_ROLE_ID:
            staff_role_ids.add(SUPPORT_STAFF_ROLE_ID)
        is_staff = has_any_role(interaction.user, staff_role_ids)

        if not (is_owner or is_staff or interaction.user.guild_permissions.manage_channels):
            await interaction.response.send_message("You do not have permission to close this ticket.", ephemeral=True)
            return

        await interaction.response.send_message("Closing ticket and saving transcript...", ephemeral=True)

        logs_channel = interaction.guild.get_channel(STAFF_LOGS_CHANNEL_ID)
        ticket_type_key = parse_topic_value(channel.topic, "ticket_type") or "support"
        ticket_type = TICKET_TYPES.get(ticket_type_key, TICKET_TYPES["support"])

        transcript_file = await export_transcript(channel)

        if isinstance(logs_channel, discord.TextChannel):
            embed = discord.Embed(
                title="🧾 Ticket Closed",
                description=(
                    f"Closed By: {interaction.user.mention}\n"
                    f"Channel: `{channel.name}`\n"
                    f"Type: {ticket_type.label}\n"
                    f"Closed At: {format_dt(utc_now())}"
                ),
                color=discord.Color.red(),
                timestamp=utc_now(),
            )
            owner_display = parse_topic_value(channel.topic, "ticket_owner")
            if owner_display:
                embed.add_field(name="Ticket Owner ID", value=owner_display, inline=False)
            apply_branding(embed)
            await logs_channel.send(embed=embed, file=transcript_file)

        await asyncio.sleep(1.5)
        await channel.delete(reason=f"Ticket closed by {interaction.user} ({interaction.user.id})")


class ApplicationReviewView(discord.ui.View):
    def __init__(self) -> None:
        super().__init__(timeout=None)

    async def _check_permission(self, interaction: discord.Interaction) -> bool:
        if not interaction.guild or not isinstance(interaction.user, discord.Member):
            return False
        allowed_roles = {LEADER_ROLE_ID, CO_LEADER_ROLE_ID, MANAGER_ROLE_ID}
        return has_any_role(interaction.user, allowed_roles) or interaction.user.guild_permissions.manage_guild

    @discord.ui.button(label="Accept", emoji="✅", style=discord.ButtonStyle.success, custom_id="diff_app_accept")
    async def accept_application(self, interaction: discord.Interaction, button: discord.ui.Button) -> None:
        if not await self._check_permission(interaction):
            await interaction.response.send_message("Only DIFF leadership can approve applications.", ephemeral=True)
            return
        if not interaction.guild or not isinstance(interaction.channel, discord.TextChannel) or not isinstance(interaction.user, discord.Member):
            await interaction.response.send_message("This action can only be used in the server.", ephemeral=True)
            return

        applicant_id_raw = parse_topic_value(interaction.channel.topic, "ticket_owner")
        if not applicant_id_raw or not applicant_id_raw.isdigit():
            await interaction.response.send_message("Could not detect the application owner from this ticket.", ephemeral=True)
            return

        applicant = interaction.guild.get_member(int(applicant_id_raw))
        if applicant is None:
            await interaction.response.send_message("The applicant is no longer in the server.", ephemeral=True)
            return

        role = interaction.guild.get_role(APPROVED_STAFF_ROLE_ID) if APPROVED_STAFF_ROLE_ID else None
        if role:
            try:
                await applicant.add_roles(role, reason=f"Application approved by {interaction.user}")
            except discord.HTTPException:
                pass

        embed = build_decision_embed(applicant, interaction.user, approved=True)
        await interaction.response.send_message("Application approved.", ephemeral=True)
        await interaction.channel.send(embed=embed)

        logs_channel = interaction.guild.get_channel(STAFF_LOGS_CHANNEL_ID)
        if isinstance(logs_channel, discord.TextChannel):
            await logs_channel.send(embed=embed)

        for child in self.children:
            child.disabled = True
        await interaction.message.edit(view=self)

    @discord.ui.button(label="Deny", emoji="❌", style=discord.ButtonStyle.danger, custom_id="diff_app_deny")
    async def deny_application(self, interaction: discord.Interaction, button: discord.ui.Button) -> None:
        if not await self._check_permission(interaction):
            await interaction.response.send_message("Only DIFF leadership can deny applications.", ephemeral=True)
            return
        if not interaction.guild or not isinstance(interaction.channel, discord.TextChannel) or not isinstance(interaction.user, discord.Member):
            await interaction.response.send_message("This action can only be used in the server.", ephemeral=True)
            return

        applicant_id_raw = parse_topic_value(interaction.channel.topic, "ticket_owner")
        if not applicant_id_raw or not applicant_id_raw.isdigit():
            await interaction.response.send_message("Could not detect the application owner from this ticket.", ephemeral=True)
            return

        applicant = interaction.guild.get_member(int(applicant_id_raw))
        if applicant is None:
            await interaction.response.send_message("The applicant is no longer in the server.", ephemeral=True)
            return

        embed = build_decision_embed(applicant, interaction.user, approved=False)
        await interaction.response.send_message("Application denied.", ephemeral=True)
        await interaction.channel.send(embed=embed)

        logs_channel = interaction.guild.get_channel(STAFF_LOGS_CHANNEL_ID)
        if isinstance(logs_channel, discord.TextChannel):
            await logs_channel.send(embed=embed)

        for child in self.children:
            child.disabled = True
        await interaction.message.edit(view=self)


class SupportDropdown(discord.ui.Select):
    def __init__(self) -> None:
        options = [
            discord.SelectOption(
                label=ticket.label,
                value=ticket.key,
                description=ticket.description[:100],
                emoji=ticket.emoji,
            )
            for ticket in TICKET_TYPES.values()
        ]
        super().__init__(
            placeholder="Choose the support option that best fits your situation...",
            min_values=1,
            max_values=1,
            options=options,
            custom_id="diff_support_dropdown_select",
        )

    async def callback(self, interaction: discord.Interaction) -> None:
        if not interaction.guild or not isinstance(interaction.user, discord.Member):
            await interaction.response.send_message("This dropdown can only be used inside the server.", ephemeral=True)
            return

        ticket = TICKET_TYPES[self.values[0]]
        existing = await find_existing_ticket(interaction.guild, interaction.user, ticket)
        if existing:
            await interaction.response.send_message(
                f"You already have an open {ticket.label} ticket: {existing.mention}",
                ephemeral=True,
            )
            return

        panel_channel = interaction.guild.get_channel(SUPPORT_PANEL_CHANNEL_ID)
        category = None

        if TICKET_CATEGORY_ID:
            category = interaction.guild.get_channel(TICKET_CATEGORY_ID)
        elif isinstance(panel_channel, discord.TextChannel):
            category = panel_channel.category

        if not isinstance(category, discord.CategoryChannel):
            await interaction.response.send_message(
                "Ticket category is not configured correctly. Set TICKET_CATEGORY_ID or place the panel inside a category.",
                ephemeral=True,
            )
            return

        staff_role_ids = {LEADER_ROLE_ID, CO_LEADER_ROLE_ID, MANAGER_ROLE_ID, HOST_ROLE_ID}
        if SUPPORT_STAFF_ROLE_ID:
            staff_role_ids.add(SUPPORT_STAFF_ROLE_ID)

        overwrites: dict[discord.abc.Snowflake, discord.PermissionOverwrite] = {
            interaction.guild.default_role: discord.PermissionOverwrite(view_channel=False),
            interaction.user: discord.PermissionOverwrite(
                view_channel=True,
                send_messages=True,
                read_message_history=True,
                attach_files=True,
                embed_links=True,
            ),
        }

        me = interaction.guild.me
        if me:
            overwrites[me] = discord.PermissionOverwrite(
                view_channel=True,
                send_messages=True,
                read_message_history=True,
                manage_channels=True,
                manage_messages=True,
                attach_files=True,
                embed_links=True,
            )

        for role_id in staff_role_ids:
            role = interaction.guild.get_role(role_id)
            if role:
                overwrites[role] = discord.PermissionOverwrite(
                    view_channel=True,
                    send_messages=True,
                    read_message_history=True,
                    manage_messages=True,
                    attach_files=True,
                    embed_links=True,
                )

        channel_name = f"{ticket.channel_prefix}-{clean_name(interaction.user.name)}"
        topic = f"ticket_owner={interaction.user.id} | ticket_type={ticket.key}"

        channel = await interaction.guild.create_text_channel(
            name=channel_name,
            category=category,
            overwrites=overwrites,
            topic=topic,
            reason=f"{ticket.title} opened by {interaction.user} ({interaction.user.id})",
        )

        ping_text = " ".join(filter(None, [interaction.user.mention, role_mention(ticket.ping_role_id)]))

        await channel.send(
            content=ping_text or None,
            embed=build_ticket_embed(ticket, interaction.user),
            view=CloseTicketView(),
        )

        if ticket.key == "apply":
            await channel.send(embed=build_application_questions_embed(interaction.user))
            await channel.send(embed=build_application_review_embed(interaction.user), view=ApplicationReviewView())

        logs_channel = interaction.guild.get_channel(STAFF_LOGS_CHANNEL_ID)
        if isinstance(logs_channel, discord.TextChannel):
            await logs_channel.send(embed=build_log_embed("Opened", interaction.user, ticket, channel))

        await interaction.response.send_message(
            f"Your {ticket.label} ticket has been created: {channel.mention}",
            ephemeral=True,
        )

class DiffSupportSystemUpgrade(commands.Cog):
    def __init__(self, bot: commands.Bot) -> None:
        self.bot = bot

    async def cog_load(self) -> None:
        self.bot.add_view(SupportDropdownView())
        self.bot.add_view(CloseTicketView())
        self.bot.add_view(ApplicationReviewView())

    @app_commands.command(name="post_support_panel", description="Post the DIFF Support Center support ticket panel.")
    async def post_support_panel(self, interaction: discord.Interaction) -> None:
        if not interaction.guild or not isinstance(interaction.user, discord.Member):
            await interaction.response.send_message("This command can only be used inside the server.", ephemeral=True)
            return

        allowed_roles = {LEADER_ROLE_ID, CO_LEADER_ROLE_ID, MANAGER_ROLE_ID}
        if not (has_any_role(interaction.user, allowed_roles) or interaction.user.guild_permissions.manage_guild):
            await interaction.response.send_message("Only DIFF leadership can post this panel.", ephemeral=True)
            return

        channel = interaction.guild.get_channel(SUPPORT_PANEL_CHANNEL_ID)
        if not isinstance(channel, discord.TextChannel):
            await interaction.response.send_message("Support panel channel was not found.", ephemeral=True)
            return

        if DELETE_OLD_PANELS_ON_REPOST:
            try:
                async for msg in channel.history(limit=50):
                    if msg.author.id == self.bot.user.id and any(embed.title == BRAND_NAME for embed in msg.embeds):
                        try:
                            await msg.delete()
                        except discord.HTTPException:
                            pass
            except discord.HTTPException:
                pass

        await channel.send(embed=build_panel_embed(), view=SupportDropdownView())
        await interaction.response.send_message(f"Support panel posted in {channel.mention}.", ephemeral=True)


async def setup(bot: commands.Bot) -> None:
    await bot.add_cog(DiffSupportSystemUpgrade(bot))
