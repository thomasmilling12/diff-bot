import json
import os
from datetime import datetime
from zoneinfo import ZoneInfo

import discord
from discord.ext import commands

# =========================================================
# DIFF Ticket -> Application Review Bridge
# =========================================================
#
# What this file does:
# - Connects your application ticket flow to review actions
# - Adds button-based staff actions inside the ticket
# - Lets staff move an applicant through:
#     Applied -> Interview Scheduled -> Approved / Denied
# - Posts updates to staff logs
# - Prevents non-staff from using review buttons
#
# Commands:
#   /send_application_review_panel
#   /link_application_ticket
#   /application_status
#
# Load with:
#   await bot.load_extension("diff_ticket_application_bridge")
#
# Notes:
# - This is made to be dropped into your current bot as a new extension
# - It does NOT replace your current auto-ticket system
# - It adds the review bridge on top of it
# =========================================================

GUILD_ID = 850386896509337710
TIMEZONE = "America/New_York"

# Channels
STAFF_LOGS_CHANNEL_ID = 1485265848099799163

# Staff roles allowed to use the buttons
ALLOWED_STAFF_ROLE_IDS = {
    850391095845584937,   # Leader
    850391378559238235,   # Co-Leader
    990011447193006101,   # Manager
}

STATE_FILE = "diff_ticket_application_bridge_state.json"


def now_et():
    return datetime.now(ZoneInfo(TIMEZONE)).strftime("%Y-%m-%d %I:%M %p ET")


def load_state():
    if not os.path.exists(STATE_FILE):
        return {"applications": {}, "ticket_links": {}}
    try:
        with open(STATE_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    except Exception:
        return {"applications": {}, "ticket_links": {}}


def save_state(state: dict):
    with open(STATE_FILE, "w", encoding="utf-8") as f:
        json.dump(state, f, indent=2)


class InterviewScheduleModal(discord.ui.Modal, title="Schedule Interview"):
    interview_time = discord.ui.TextInput(
        label="Interview time",
        placeholder="Example: Friday 7:30 PM ET",
        max_length=100,
        required=True,
    )
    notes = discord.ui.TextInput(
        label="Interview notes",
        placeholder="Any notes for staff or the applicant",
        style=discord.TextStyle.paragraph,
        required=False,
        max_length=500,
    )

    def __init__(self, cog, target_member_id: int):
        super().__init__()
        self.cog = cog
        self.target_member_id = target_member_id

    async def on_submit(self, interaction: discord.Interaction):
        member = interaction.guild.get_member(self.target_member_id)
        if member is None:
            await interaction.response.send_message("Applicant not found in this server.", ephemeral=True)
            return

        app = self.cog.get_app(member.id)
        app["display_name"] = member.display_name
        app["status"] = "Interview Scheduled"
        app["interview_scheduled_for"] = str(self.interview_time)
        app["interview_notes"] = str(self.notes) if self.notes else "No notes provided."
        app["reviewed_by"] = interaction.user.mention
        if not app.get("submitted_at"):
            app["submitted_at"] = now_et()
        app["last_updated_at"] = now_et()
        save_state(self.cog.state)

        await self.cog.post_staff_log(
            title="🎤 Interview Scheduled",
            description=(
                f"**Applicant:** {member.mention}\n"
                f"**Interview:** {self.interview_time}\n"
                f"**Notes:** {self.notes or 'No notes provided.'}\n"
                f"**Reviewed By:** {interaction.user.mention}"
            ),
            color=discord.Color.orange(),
        )

        await self.cog.update_ticket_panel(interaction.channel, member)
        await interaction.response.send_message(
            f"✅ Interview scheduled for {member.mention}.",
            ephemeral=True,
        )


class ResultModal(discord.ui.Modal):
    result_notes = discord.ui.TextInput(
        label="Result notes",
        placeholder="Reason or final notes",
        style=discord.TextStyle.paragraph,
        required=False,
        max_length=500,
    )

    def __init__(self, cog, target_member_id: int, result_name: str):
        self.cog = cog
        self.target_member_id = target_member_id
        self.result_name = result_name
        super().__init__(title=f"{result_name} Application")

    async def on_submit(self, interaction: discord.Interaction):
        member = interaction.guild.get_member(self.target_member_id)
        if member is None:
            await interaction.response.send_message("Applicant not found in this server.", ephemeral=True)
            return

        app = self.cog.get_app(member.id)
        app["display_name"] = member.display_name
        app["status"] = self.result_name
        app["result_notes"] = str(self.result_notes) if self.result_notes else "No notes provided."
        app["reviewed_by"] = interaction.user.mention
        if not app.get("submitted_at"):
            app["submitted_at"] = now_et()
        app["last_updated_at"] = now_et()
        save_state(self.cog.state)

        color = discord.Color.green() if self.result_name == "Approved" else discord.Color.red()

        await self.cog.post_staff_log(
            title="✅ Application Approved" if self.result_name == "Approved" else "❌ Application Denied",
            description=(
                f"**Applicant:** {member.mention}\n"
                f"**Status:** {self.result_name}\n"
                f"**Notes:** {self.result_notes or 'No notes provided.'}\n"
                f"**Reviewed By:** {interaction.user.mention}"
            ),
            color=color,
        )

        await self.cog.update_ticket_panel(interaction.channel, member)
        await interaction.response.send_message(
            f"✅ Application marked as **{self.result_name}** for {member.mention}.",
            ephemeral=True,
        )


class ApplicationReviewView(discord.ui.View):
    def __init__(self, cog, target_member_id: int):
        super().__init__(timeout=None)
        self.cog = cog
        self.target_member_id = target_member_id

    async def interaction_check(self, interaction: discord.Interaction) -> bool:
        if not self.cog.is_staff(interaction.user):
            await interaction.response.send_message(
                "You do not have permission to use these review buttons.",
                ephemeral=True,
            )
            return False
        return True

    @discord.ui.button(label="Mark Applied", style=discord.ButtonStyle.secondary, emoji="🧾")
    async def mark_applied(self, interaction: discord.Interaction, button: discord.ui.Button):
        member = interaction.guild.get_member(self.target_member_id)
        if member is None:
            await interaction.response.send_message("Applicant not found in this server.", ephemeral=True)
            return

        app = self.cog.get_app(member.id)
        app["display_name"] = member.display_name
        app["status"] = "Applied"
        app["submitted_at"] = app.get("submitted_at") or now_et()
        app["reviewed_by"] = interaction.user.mention
        app["last_updated_at"] = now_et()
        save_state(self.cog.state)

        await self.cog.post_staff_log(
            title="🧾 Application Linked",
            description=(
                f"**Applicant:** {member.mention}\n"
                f"**Status:** Applied\n"
                f"**Linked By:** {interaction.user.mention}\n"
                f"**Ticket:** {interaction.channel.mention}"
            ),
            color=discord.Color.blurple(),
        )

        await self.cog.update_ticket_panel(interaction.channel, member)
        await interaction.response.send_message(f"✅ {member.mention} marked as applied.", ephemeral=True)

    @discord.ui.button(label="Schedule Interview", style=discord.ButtonStyle.primary, emoji="🎤")
    async def schedule_interview(self, interaction: discord.Interaction, button: discord.ui.Button):
        await interaction.response.send_modal(
            InterviewScheduleModal(self.cog, self.target_member_id)
        )

    @discord.ui.button(label="Approve", style=discord.ButtonStyle.success, emoji="✅")
    async def approve(self, interaction: discord.Interaction, button: discord.ui.Button):
        await interaction.response.send_modal(
            ResultModal(self.cog, self.target_member_id, "Approved")
        )

    @discord.ui.button(label="Deny", style=discord.ButtonStyle.danger, emoji="❌")
    async def deny(self, interaction: discord.Interaction, button: discord.ui.Button):
        await interaction.response.send_modal(
            ResultModal(self.cog, self.target_member_id, "Denied")
        )


class DiffTicketApplicationBridge(commands.Cog):
    def __init__(self, bot: commands.Bot):
        self.bot = bot
        self.state = load_state()

    # -----------------------------
    # helpers
    # -----------------------------
    def is_staff(self, member: discord.Member) -> bool:
        return any(role.id in ALLOWED_STAFF_ROLE_IDS for role in member.roles)

    def get_guild(self):
        return self.bot.get_guild(GUILD_ID)

    def get_app(self, user_id: int):
        key = str(user_id)
        apps = self.state["applications"]
        if key not in apps:
            apps[key] = {
                "user_id": user_id,
                "display_name": None,
                "status": "Applied",
                "submitted_at": None,
                "interview_scheduled_for": None,
                "interview_notes": None,
                "result_notes": None,
                "reviewed_by": None,
                "last_updated_at": None,
            }
        return apps[key]

    def build_status_embed(self, member: discord.Member):
        app = self.get_app(member.id)

        embed = discord.Embed(
            title="📋 Application Review Panel",
            description=(
                f"Use the buttons below to move **{member.mention}** through the application process.\n\n"
                "This ticket is now directly connected to application review and interview actions."
            ),
            color=discord.Color.blue(),
            timestamp=datetime.utcnow(),
        )

        embed.add_field(name="Applicant", value=member.mention, inline=False)
        embed.add_field(name="Current Status", value=f"**{app.get('status', 'Unknown')}**", inline=True)
        embed.add_field(name="Submitted", value=app.get("submitted_at") or "Not logged", inline=True)

        if app.get("interview_scheduled_for"):
            embed.add_field(
                name="Interview Scheduled",
                value=app["interview_scheduled_for"],
                inline=False,
            )
        if app.get("interview_notes"):
            embed.add_field(
                name="Interview Notes",
                value=app["interview_notes"],
                inline=False,
            )
        if app.get("result_notes"):
            embed.add_field(
                name="Result Notes",
                value=app["result_notes"],
                inline=False,
            )
        if app.get("reviewed_by"):
            embed.add_field(
                name="Last Reviewed By",
                value=app["reviewed_by"],
                inline=False,
            )

        embed.set_footer(text="Different Meets • Ticket Application Bridge")
        return embed

    async def post_staff_log(self, title: str, description: str, color: discord.Color):
        guild = self.get_guild()
        if guild is None:
            return

        channel = guild.get_channel(STAFF_LOGS_CHANNEL_ID)
        if not isinstance(channel, discord.TextChannel):
            return

        embed = discord.Embed(
            title=title,
            description=description,
            color=color,
            timestamp=datetime.utcnow(),
        )
        embed.set_footer(text="Different Meets • Staff Logs")
        await channel.send(embed=embed)

    async def update_ticket_panel(self, channel: discord.TextChannel, member: discord.Member):
        ticket_key = str(channel.id)
        message_id = self.state["ticket_links"].get(ticket_key, {}).get("panel_message_id")

        embed = self.build_status_embed(member)
        view = ApplicationReviewView(self, member.id)

        if message_id:
            try:
                msg = await channel.fetch_message(message_id)
                await msg.edit(embed=embed, view=view)
                save_state(self.state)
                return
            except Exception:
                pass

        msg = await channel.send(embed=embed, view=view)
        self.state["ticket_links"][ticket_key] = {
            "member_id": member.id,
            "panel_message_id": msg.id,
        }
        save_state(self.state)

    # -----------------------------
    # commands
    # -----------------------------
    @commands.hybrid_command(
        name="send_application_review_panel",
        description="Post the application review panel in the current ticket."
    )
    @commands.has_permissions(manage_guild=True)
    async def send_application_review_panel(self, ctx: commands.Context, member: discord.Member):
        await self.update_ticket_panel(ctx.channel, member)
        await ctx.reply(
            f"✅ Application review panel connected to {member.mention} in this ticket.",
            mention_author=False,
        )

    @commands.hybrid_command(
        name="link_application_ticket",
        description="Link the current ticket channel to an applicant."
    )
    @commands.has_permissions(manage_guild=True)
    async def link_application_ticket(self, ctx: commands.Context, member: discord.Member):
        app = self.get_app(member.id)
        app["display_name"] = member.display_name
        app["submitted_at"] = app.get("submitted_at") or now_et()
        app["status"] = app.get("status") or "Applied"
        app["last_updated_at"] = now_et()
        app["reviewed_by"] = ctx.author.mention

        self.state["ticket_links"][str(ctx.channel.id)] = {
            "member_id": member.id,
        }
        save_state(self.state)

        await self.update_ticket_panel(ctx.channel, member)
        await self.post_staff_log(
            title="🔗 Ticket Linked to Application",
            description=(
                f"**Applicant:** {member.mention}\n"
                f"**Ticket:** {ctx.channel.mention}\n"
                f"**Linked By:** {ctx.author.mention}"
            ),
            color=discord.Color.blurple(),
        )
        await ctx.reply(
            f"✅ This ticket is now linked to {member.mention}'s application.",
            mention_author=False,
        )

    @commands.hybrid_command(
        name="application_status",
        description="Show the linked application status for this ticket."
    )
    @commands.has_permissions(manage_guild=True)
    async def application_status(self, ctx: commands.Context):
        link = self.state["ticket_links"].get(str(ctx.channel.id))
        if not link:
            await ctx.reply("This ticket is not linked to an application yet.", mention_author=False)
            return

        member = ctx.guild.get_member(link["member_id"])
        if member is None:
            await ctx.reply("That linked applicant is no longer in the server.", mention_author=False)
            return

        embed = self.build_status_embed(member)
        await ctx.reply(embed=embed, mention_author=False)


async def setup(bot: commands.Bot):
    await bot.add_cog(DiffTicketApplicationBridge(bot))
