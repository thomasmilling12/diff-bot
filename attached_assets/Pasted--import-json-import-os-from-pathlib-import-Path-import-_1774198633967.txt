def load_json(path: str, default):
    try:
        if os.path.exists(path):
            with open(path, "r", encoding="utf-8") as f:
                return json.load(f)
    except Exception:
        pass
    return default


def save_json(path: str, data):
    with open(path, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=2)

def build_member_stats_embed(member: discord.Member, stats: dict) -> discord.Embed:
    attended = stats.get("attended", 0)
    hosted = stats.get("hosted", 0)
    no_shows = stats.get("no_shows", 0)
    strikes = stats.get("strikes", 0)
    warnings = stats.get("warnings", 0)

    score = max(0, (attended * 5) + (hosted * 8) - (no_shows * 6) - (strikes * 10) - (warnings * 3))
    if score >= 80:
        grade = "A"
    elif score >= 60:
        grade = "B"
    elif score >= 40:
        grade = "C"
    else:
        grade = "D"

    embed = discord.Embed(
        title=f"📊 {member.display_name} — DIFF Member Stats",
        description="Your current DIFF activity snapshot.",
        color=discord.Color.blue()
    )
    embed.add_field(name="✅ Meets Attended", value=str(attended), inline=True)
    embed.add_field(name="🎤 Meets Hosted", value=str(hosted), inline=True)
    embed.add_field(name="❌ No-Shows", value=str(no_shows), inline=True)
    embed.add_field(name="⚠️ Warnings", value=str(warnings), inline=True)
    embed.add_field(name="🚨 Strikes", value=str(strikes), inline=True)
    embed.add_field(name="🏅 Activity Grade", value=f"{grade} ({score})", inline=True)
    embed.set_footer(text="Stats update when your activity systems are used.")
    return embed

# COG
# =========================
class DiffElitePanelCog(commands.Cog):
    def __init__(self, bot: commands.Bot):
        self.bot = bot
        self.panel_state = load_json(PANEL_STATE_FILE, {})
        self.bot.add_view(PanelView(self))

    def get_panel_state(self) -> dict:
        return load_json(PANEL_STATE_FILE, {})

    def save_panel_state(self, data: dict):
        save_json(PANEL_STATE_FILE, data)
        self.panel_state = data

    async def post_or_refresh_panel(self, channel: discord.TextChannel):
        state = self.get_panel_state()
        guild_key = str(channel.guild.id)
        guild_state = state.get(guild_key, {})
        message_id = guild_state.get("message_id")
        channel_id = guild_state.get("channel_id")

        embed = build_main_embed()
        view = PanelView(self)

        target_message = None

        if message_id and channel_id:
            existing_channel = channel.guild.get_channel(int(channel_id))
            if existing_channel:
                try:
                    target_message = await existing_channel.fetch_message(int(message_id))
                except discord.NotFound:
                    target_message = None
                except discord.HTTPException:
                    target_message = None

        if target_message:
            await target_message.edit(embed=embed, view=view)
            return target_message, False

        posted = await channel.send(embed=embed, view=view)
        state[guild_key] = {
            "channel_id": str(channel.id),
            "message_id": str(posted.id),
        }
        self.save_panel_state(state)
        return posted, True

    @app_commands.command(name="diffpanel", description="Post or refresh the DIFF control panel without duplicates")
    async def diffpanel(self, interaction: discord.Interaction):
        msg, created = await self.post_or_refresh_panel(interaction.channel)
        if created:
            await interaction.response.send_message("DIFF control panel posted ✅", ephemeral=True)
        else:
            await interaction.response.send_message("DIFF control panel refreshed ✅", ephemeral=True)

    @app_commands.command(name="diffpanel-refresh", description="Force refresh the saved DIFF control panel")
    async def diffpanel_refresh(self, interaction: discord.Interaction):
        msg, created = await self.post_or_refresh_panel(interaction.channel)
        await interaction.response.send_message("DIFF control panel refreshed ✅", ephemeral=True)

    @app_commands.command(name="diffpanel-info", description="View quick info for the DIFF panel systems")
    async def diffpanel_info(self, interaction: discord.Interaction):
        embed = discord.Embed(
            title="🚗 DIFF Panel System Guide",
            description=(
                "**About button descriptions:** Discord buttons do not support true hover tooltips.\n"
                "This file uses clear labels, matching icons, and this info command instead.\n\n"
                "📝 **Crew Roll Call** — Attendance and activity check\n"
                "🎨 **Crew Color Voting** — Weekly color/theme voting\n"
                "⚠️ **Strike System** — Conduct rules and standing\n"
                "🧥 **Crew Jackets** — Outfit guidance\n"
                "📊 **Member Stats** — Personal activity snapshot"
            ),
            color=discord.Color.blurple()
        )
        await interaction.response.send_message(embed=embed, ephemeral=True)


async def setup(bot):
    await bot.add_cog(DiffElitePanelCog(bot))
