"""
DIFF Real Attendance System
Replit-ready discord.py extension/module

What this file does
- Auto-posts a real attendance panel in your target channel
- Refreshes the same panel without duplicates
- Staff can create attendance sessions for an event/meet
- Members press Check In when the meet starts
- Bot marks real attendance
- Bot tracks no-shows by comparing RSVP list vs actual check-ins
- Staff can close attendance and get a summary
- Optional log channel support
- Persistent buttons survive bot restarts

How to use
1) Put this file in your Replit project
2) Fill in the IDs below
3) Load the extension in your main bot file:
       await bot.load_extension("attendance_system")

Requirements
- discord.py 2.x

Note
- This can work standalone
- If you want, you can later connect it to your crew_events_system JSON for deeper integration
"""

import json
import os
from datetime import datetime, timezone
from typing import Optional

import discord
from discord.ext import commands

# =========================================================
# CONFIG
# =========================================================
TARGET_CHANNEL_ID = 0        # attendance / crew-events / meet-starts channel
STAFF_ROLE_ID = 0            # staff role allowed to create/close attendance sessions
LOG_CHANNEL_ID = 0           # optional log channel, 0 disables logs
ATTENDANCE_PING_ROLE_ID = 0  # optional role to ping when attendance opens, 0 disables

DATA_FILE = "attendance_panel_data.json"
ATTENDANCE_FILE = "attendance_sessions.json"

EMBED_COLOR = 0x5865F2
SUCCESS_COLOR = 0x57F287
WARNING_COLOR = 0xFEE75C
ERROR_COLOR = 0xED4245

THUMBNAIL_URL = ""
BANNER_IMAGE_URL = ""


# =========================================================
# FILE HELPERS
# =========================================================
def load_json(path: str) -> dict:
    if not os.path.exists(path):
        return {}
    try:
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f)
    except Exception:
        return {}


def save_json(path: str, data: dict) -> None:
    with open(path, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=4)


def get_saved_panel_message_id() -> Optional[int]:
    data = load_json(DATA_FILE)
    value = data.get("panel_message_id")
    return int(value) if value else None


def save_panel_message_id(message_id: int) -> None:
    data = load_json(DATA_FILE)
    data["panel_message_id"] = message_id
    save_json(DATA_FILE, data)


def load_sessions() -> dict:
    data = load_json(ATTENDANCE_FILE)
    if "sessions" not in data:
        data["sessions"] = {}
    return data


def save_sessions(data: dict) -> None:
    save_json(ATTENDANCE_FILE, data)


def now_ts() -> int:
    return int(datetime.now(timezone.utc).timestamp())


def make_session_id() -> str:
    return str(int(datetime.now(timezone.utc).timestamp() * 1000))


# =========================================================
# PERMISSIONS
# =========================================================
def has_staff_access(member: discord.Member) -> bool:
    if member.guild_permissions.administrator:
        return True
    if member.guild_permissions.manage_guild or member.guild_permissions.manage_messages:
        return True
    if STAFF_ROLE_ID and any(role.id == STAFF_ROLE_ID for role in member.roles):
        return True
    return False


# =========================================================
# EMBEDS
# =========================================================
def build_session_embed(session_id: str, session: dict) -> discord.Embed:
    checked_in_count = len(session.get("checked_in_users", {}))
    expected_count = len(session.get("expected_users", {}))
    no_show_count = len(session.get("no_show_users", {}))

    status = session.get("status", "open").upper()

    embed = discord.Embed(
        title="✅ DIFF Real Attendance Session",
        color=EMBED_COLOR if session.get("status") == "open" else WARNING_COLOR,
        description=(
            f"**Session ID:** `{session_id}`\n"
            f"**Event Title:** {session.get('title', 'Untitled Meet')}\n"
            f"**Host:** {session.get('host', 'Not set')}\n"
            f"**Theme:** {session.get('theme', 'Not set')}\n"
            f"**Status:** **{status}**\n"
            f"**Started:** <t:{session.get('created_at', now_ts())}:F>"
        ),
    )

    embed.add_field(
        name="Live Tracking",
        value=(
            f"✅ Checked In: **{checked_in_count}**\n"
            f"📋 Expected (RSVP): **{expected_count}**\n"
            f"❌ No-Shows: **{no_show_count}**"
        ),
        inline=False,
    )

    embed.add_field(
        name="How it Works",
        value=(
            "Members who actually show up should press **Check In**.\n"
            "When staff closes this attendance session, the system compares "
            "expected members vs real check-ins and flags no-shows automatically."
        ),
        inline=False,
    )

    if session.get("notes"):
        embed.add_field(name="Notes", value=session["notes"], inline=False)

    embed.set_footer(text="DIFF Real Attendance System")
    return embed


def build_panel_embed() -> discord.Embed:
    embed = discord.Embed(
        title="🧠 DIFF Real Attendance System",
        color=EMBED_COLOR,
        description=(
            "This system tracks **real attendance**, not just RSVPs.\n\n"
            "**Use it to:**\n"
            "• Open a check-in session when the meet starts\n"
            "• Let real members check in with a button\n"
            "• Auto track who actually showed up\n"
            "• Flag no-shows when the session closes"
        ),
    )

    embed.add_field(
        name="Staff Flow",
        value=(
            "1. Press **Create Attendance Session**\n"
            "2. Enter event title, host, theme, and optional RSVP list\n"
            "3. Members press **Check In** on the session post\n"
            "4. Staff closes session and reviews no-shows"
        ),
        inline=False,
    )

    embed.add_field(
        name="Best Use",
        value=(
            "Use this when a meet actually starts so RSVP data becomes real attendance data."
        ),
        inline=False,
    )

    embed.add_field(
        name="Anti-Dupe",
        value="This panel refreshes the same post instead of posting duplicates.",
        inline=False,
    )

    if THUMBNAIL_URL:
        embed.set_thumbnail(url=THUMBNAIL_URL)
    if BANNER_IMAGE_URL:
        embed.set_image(url=BANNER_IMAGE_URL)

    embed.set_footer(text="DIFF Attendance Control")
    return embed

class CreateAttendanceSessionModal(discord.ui.Modal, title="Create Attendance Session"):
    title_input = discord.ui.TextInput(
        label="Event Title",
        placeholder="Example: Friday Night Meet",
        required=True,
        max_length=100,
    )
    host_input = discord.ui.TextInput(
        label="Host",
        placeholder="@Host or host name",
        required=True,
        max_length=100,
    )
    theme_input = discord.ui.TextInput(
        label="Theme",
        placeholder="Example: Clean JDM / Pop-Up / Crew Collab",
        required=True,
        max_length=100,
    )
    expected_input = discord.ui.TextInput(
        label="Expected User IDs (optional, comma separated)",
        placeholder="123,456,789 or leave blank if no RSVP list",
        required=False,
        style=discord.TextStyle.paragraph,
        max_length=500,
    )
    notes_input = discord.ui.TextInput(
        label="Notes (optional)",
        placeholder="Any notes for this attendance session",
        required=False,
        style=discord.TextStyle.paragraph,
        max_length=400,
    )

    def __init__(self, cog: "AttendanceCog"):
        super().__init__()
        self.cog = cog

    async def on_submit(self, interaction: discord.Interaction):
        member = interaction.user if isinstance(interaction.user, discord.Member) else None
        if member is None or not has_staff_access(member):
            return await interaction.response.send_message(
                "Only staff can create attendance sessions.",
                ephemeral=True,
            )

        expected_users = {}
        raw_ids = str(self.expected_input).strip()
        if raw_ids:
            chunks = raw_ids.replace("\n", ",").split(",")
            for chunk in chunks:
                uid = chunk.strip().replace("<@", "").replace(">", "").replace("!", "")
                if uid.isdigit():
                    expected_users[uid] = True

        session_id = make_session_id()
        data = load_sessions()
        data["sessions"][session_id] = {
            "title": str(self.title_input),
            "host": str(self.host_input),
            "theme": str(self.theme_input),
            "notes": str(self.notes_input).strip(),
            "status": "open",
            "channel_id": interaction.channel_id,
            "message_id": None,
            "expected_users": expected_users,
            "checked_in_users": {},
            "no_show_users": {},
            "created_by": interaction.user.id,
            "created_at": now_ts(),
            "closed_at": None,
        }
        save_sessions(data)

        embed = build_session_embed(session_id, data["sessions"][session_id])
        view = AttendanceSessionView(self.cog, session_id)

        content = None
        if ATTENDANCE_PING_ROLE_ID:
            role = interaction.guild.get_role(ATTENDANCE_PING_ROLE_ID) if interaction.guild else None
            if role:
                content = role.mention

        await interaction.response.send_message(
            content=content,
            embed=embed,
            view=view,
            allowed_mentions=discord.AllowedMentions(roles=True),
        )

        sent = await interaction.original_response()
        data = load_sessions()
        if session_id in data["sessions"]:
            data["sessions"][session_id]["message_id"] = sent.id
            save_sessions(data)

        await self.cog.log_action(
            interaction.guild,
            f"✅ Attendance session `{session_id}` created in <#{interaction.channel_id}> by {interaction.user.mention}"
        )

class AttendanceSessionView(discord.ui.View):
    def __init__(self, cog: "AttendanceCog", session_id: str):
        super().__init__(timeout=None)
        self.cog = cog
        self.session_id = session_id

    async def refresh_message(self, interaction: discord.Interaction, session: dict):
        try:
            embed = build_session_embed(self.session_id, session)
            await interaction.message.edit(embed=embed, view=self)
        except Exception:
            pass

    @discord.ui.button(
        label="Check In",
        emoji="✅",
        style=discord.ButtonStyle.success,
        custom_id="diff_attendance_check_in_v1",
    )
    async def check_in(self, interaction: discord.Interaction, button: discord.ui.Button):
        data = load_sessions()
        session = data.get("sessions", {}).get(self.session_id)

        if not session:
            return await interaction.response.send_message(
                "This attendance session could not be found.",
                ephemeral=True,
            )

        if session.get("status") != "open":
            return await interaction.response.send_message(
                "This attendance session is already closed.",
                ephemeral=True,
            )

        user_id = str(interaction.user.id)
        if user_id in session["checked_in_users"]:
            return await interaction.response.send_message(
                "You are already checked in for this session.",
                ephemeral=True,
            )

        session["checked_in_users"][user_id] = {
            "username": str(interaction.user),
            "timestamp": now_ts(),
        }

        # If they were previously marked no-show somehow, remove it
        session["no_show_users"].pop(user_id, None)

        save_sessions(data)
        await self.refresh_message(interaction, session)

        await interaction.response.send_message(
            "You have been checked in successfully. ✅",
            ephemeral=True,
        )

    @discord.ui.button(
        label="Close Attendance",
        emoji="🔒",
        style=discord.ButtonStyle.danger,
        custom_id="diff_attendance_close_v1",
    )
    async def close_attendance(self, interaction: discord.Interaction, button: discord.ui.Button):
        member = interaction.user if isinstance(interaction.user, discord.Member) else None
        if member is None or not has_staff_access(member):
            return await interaction.response.send_message(
                "Only staff can close attendance sessions.",
                ephemeral=True,
            )

        data = load_sessions()
        session = data.get("sessions", {}).get(self.session_id)

        if not session:
            return await interaction.response.send_message(
                "This attendance session could not be found.",
                ephemeral=True,
            )

        if session.get("status") == "closed":
            return await interaction.response.send_message(
                "This attendance session is already closed.",
                ephemeral=True,
            )

        expected_users = set(session.get("expected_users", {}).keys())
        checked_in_users = set(session.get("checked_in_users", {}).keys())

        no_shows = {}
        for uid in expected_users:
            if uid not in checked_in_users:
                no_shows[uid] = True

        session["no_show_users"] = no_shows
        session["status"] = "closed"
        session["closed_at"] = now_ts()
        session["closed_by"] = interaction.user.id

        save_sessions(data)

        embed = build_session_embed(self.session_id, session)
        try:
            await interaction.message.edit(embed=embed, view=self)
        except Exception:
            pass

        checked_in_count = len(session.get("checked_in_users", {}))
        expected_count = len(session.get("expected_users", {}))
        no_show_count = len(session.get("no_show_users", {}))

        summary = discord.Embed(
            title="📊 Attendance Session Closed",
            color=WARNING_COLOR,
            description=(
                f"**Event:** {session.get('title', 'Untitled Meet')}\n"
                f"**Checked In:** **{checked_in_count}**\n"
                f"**Expected:** **{expected_count}**\n"
                f"**No-Shows:** **{no_show_count}**"
            ),
        )

        if no_show_count:
            sample = []
            for uid in list(session["no_show_users"].keys())[:15]:
                sample.append(f"<@{uid}>")
            summary.add_field(
                name="No-Show Members",
                value="\n".join(sample) if sample else "No-show list saved.",
                inline=False,
            )
        else:
            summary.add_field(
                name="No-Show Members",
                value="No no-shows were detected.",
                inline=False,
            )

        summary.set_footer(text=f"Closed by {interaction.user.display_name}")

        await interaction.response.send_message(embed=summary)

        await self.cog.log_action(
            interaction.guild,
            f"📊 Attendance session `{self.session_id}` closed by {interaction.user.mention} | "
            f"Checked In: {checked_in_count} | Expected: {expected_count} | No-Shows: {no_show_count}"
        )


class AttendancePanelView(discord.ui.View):
    def __init__(self, cog: "AttendanceCog"):
        super().__init__(timeout=None)
        self.cog = cog

    @discord.ui.button(
        label="Create Attendance Session",
        emoji="🧠",
        style=discord.ButtonStyle.primary,
        custom_id="diff_attendance_create_session_v1",
    )
    async def create_session(self, interaction: discord.Interaction, button: discord.ui.Button):
        member = interaction.user if isinstance(interaction.user, discord.Member) else None
        if member is None or not has_staff_access(member):
            return await interaction.response.send_message(
                "Only staff can create attendance sessions.",
                ephemeral=True,
            )
        await interaction.response.send_modal(CreateAttendanceSessionModal(self.cog))

    @discord.ui.button(
        label="View Open Sessions",
        emoji="📌",
        style=discord.ButtonStyle.secondary,
        custom_id="diff_attendance_view_open_v1",
    )
    async def view_open_sessions(self, interaction: discord.Interaction, button: discord.ui.Button):
        data = load_sessions()
        sessions = data.get("sessions", {})
        open_items = []

        for session_id, session in sessions.items():
            if session.get("status") == "open":
                open_items.append((session.get("created_at", 0), session_id, session))

        open_items.sort(key=lambda x: x[0], reverse=True)

        if not open_items:
            return await interaction.response.send_message(
                "There are no open attendance sessions right now.",
                ephemeral=True,
            )

        embed = discord.Embed(
            title="📌 Open Attendance Sessions",
            color=EMBED_COLOR,
            description="Here are the currently open real attendance sessions.",
        )

        for _, session_id, session in open_items[:10]:
            embed.add_field(
                name=f"{session.get('title', 'Untitled Meet')} • `{session_id}`",
                value=(
                    f"Host: {session.get('host', 'Not set')}\n"
                    f"Theme: {session.get('theme', 'Not set')}\n"
                    f"Checked In: {len(session.get('checked_in_users', {}))}\n"
                    f"Expected: {len(session.get('expected_users', {}))}"
                ),
                inline=False,
            )

        await interaction.response.send_message(embed=embed, ephemeral=True)

    @discord.ui.button(
        label="Refresh Panel",
        emoji="♻️",
        style=discord.ButtonStyle.secondary,
        custom_id="diff_attendance_refresh_panel_v1",
    )
    async def refresh_panel(self, interaction: discord.Interaction, button: discord.ui.Button):
        member = interaction.user if isinstance(interaction.user, discord.Member) else None
        if member is None or not has_staff_access(member):
            return await interaction.response.send_message(
                "Only staff can refresh this panel.",
                ephemeral=True,
            )

        await interaction.response.defer(ephemeral=True)
        await self.cog.ensure_panel()
        await interaction.followup.send("Attendance panel refreshed.", ephemeral=True)


# =========================================================
# COG
# =========================================================
class AttendanceCog(commands.Cog):
    def __init__(self, bot: commands.Bot):
        self.bot = bot
        self.panel_view = AttendancePanelView(self)
        self.bot.add_view(self.panel_view)

    async def cog_load(self):
        self.bot.add_view(self.panel_view)
        await self.rebuild_session_views()

    async def rebuild_session_views(self):
        data = load_sessions()
        for session_id in data.get("sessions", {}).keys():
            self.bot.add_view(AttendanceSessionView(self, session_id))

    async def log_action(self, guild: Optional[discord.Guild], message: str):
        if guild is None or LOG_CHANNEL_ID == 0:
            return
        channel = guild.get_channel(LOG_CHANNEL_ID)
        if channel and isinstance(channel, discord.TextChannel):
            try:
                await channel.send(message)
            except Exception:
                pass

    async def get_target_channel(self) -> Optional[discord.TextChannel]:
        channel = self.bot.get_channel(TARGET_CHANNEL_ID)
        if channel is None:
            try:
                fetched = await self.bot.fetch_channel(TARGET_CHANNEL_ID)
                if isinstance(fetched, discord.TextChannel):
                    return fetched
                return None
            except Exception:
                return None
        return channel if isinstance(channel, discord.TextChannel) else None

    async def ensure_panel(self):
        channel = await self.get_target_channel()
        if channel is None:
            print(f"[Attendance] Could not find channel: {TARGET_CHANNEL_ID}")
            return

        embed = build_panel_embed()
        saved_message_id = get_saved_panel_message_id()

        if saved_message_id:
            try:
                msg = await channel.fetch_message(saved_message_id)
                await msg.edit(embed=embed, view=self.panel_view)
                print("[Attendance] Existing panel refreshed.")
                return
            except discord.NotFound:
                pass
            except Exception as e:
                print(f"[Attendance] Failed to edit saved panel: {e}")

        try:
            async for msg in channel.history(limit=40):
                if (
                    msg.author == self.bot.user
                    and msg.embeds
                    and msg.embeds[0].title == "🧠 DIFF Real Attendance System"
                ):
                    try:
                        await msg.delete()
                    except Exception:
                        pass
        except Exception as e:
            print(f"[Attendance] Duplicate cleanup issue: {e}")

        try:
            new_msg = await channel.send(embed=embed, view=self.panel_view)
            save_panel_message_id(new_msg.id)
            print(f"[Attendance] New panel posted: {new_msg.id}")
        except Exception as e:
            print(f"[Attendance] Failed to post panel: {e}")

    @commands.Cog.listener()
    async def on_ready(self):
        await self.ensure_panel()
        await self.rebuild_session_views()

    @commands.command(name="refresh_attendance_panel")
    @commands.has_permissions(manage_guild=True)
    async def refresh_attendance_panel(self, ctx: commands.Context):
        await self.ensure_panel()
        await ctx.send("Attendance panel refreshed.", delete_after=10)


# =========================================================
# EXTENSION SETUP
# =========================================================
async def setup(bot: commands.Bot):
    await bot.add_cog(AttendanceCog(bot))
