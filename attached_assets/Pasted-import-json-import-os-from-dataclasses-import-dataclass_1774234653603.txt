import json
import os
from dataclasses import dataclass, asdict
from datetime import datetime, timezone
from typing import Dict, Optional, Set

import discord
from discord import app_commands
from discord.ext import commands

ATTENDANCE_CHANNEL_ID = 1485469927312850974
DATA_DIR = "data"
ATTENDANCE_FILE = os.path.join(DATA_DIR, "attendance_data.json")
LEADERBOARD_FILE = os.path.join(DATA_DIR, "attendance_leaderboard.json")
PROMOTION_FILE = os.path.join(DATA_DIR, "promotion_suggestions.json")

# Change these role IDs to match your server if needed.
PROMOTION_PATH = {
    "Crew Member": "Host",
    "Host": "Manager",
    "Manager": "Co-Leader",
    "Co-Leader": "Leader",
}

PROMOTION_THRESHOLDS = {
    "Crew Member": 5,
    "Host": 10,
    "Manager": 18,
    "Co-Leader": 30,
}


os.makedirs(DATA_DIR, exist_ok=True)


@dataclass
class MeetRecord:
    meet_id: str
    title: str
    host_id: int
    host_name: str
    meet_date: str
    created_at: str
    channel_id: int
    message_id: Optional[int] = None
    attendees_yes: Optional[Set[int]] = None
    attendees_maybe: Optional[Set[int]] = None
    attendees_no: Optional[Set[int]] = None
    checked_in: Optional[Set[int]] = None
    closed: bool = False

    def __post_init__(self):
        self.attendees_yes = set(self.attendees_yes or [])
        self.attendees_maybe = set(self.attendees_maybe or [])
        self.attendees_no = set(self.attendees_no or [])
        self.checked_in = set(self.checked_in or [])

    def to_json(self):
        data = asdict(self)
        data["attendees_yes"] = list(self.attendees_yes)
        data["attendees_maybe"] = list(self.attendees_maybe)
        data["attendees_no"] = list(self.attendees_no)
        data["checked_in"] = list(self.checked_in)
        return data

    @classmethod
    def from_json(cls, data):
        return cls(**data)


class AttendanceStorage:
    @staticmethod
    def _read_json(path: str, default):
        if not os.path.exists(path):
            return default
        try:
            with open(path, "r", encoding="utf-8") as f:
                return json.load(f)
        except Exception:
            return default

    @staticmethod
    def _write_json(path: str, data):
        with open(path, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=2)

    @classmethod
    def load_meets(cls) -> Dict[str, MeetRecord]:
        raw = cls._read_json(ATTENDANCE_FILE, {})
        return {meet_id: MeetRecord.from_json(data) for meet_id, data in raw.items()}

    @classmethod
    def save_meets(cls, meets: Dict[str, MeetRecord]):
        payload = {meet_id: record.to_json() for meet_id, record in meets.items()}
        cls._write_json(ATTENDANCE_FILE, payload)

    @classmethod
    def load_leaderboard(cls):
        return cls._read_json(LEADERBOARD_FILE, {})

    @classmethod
    def save_leaderboard(cls, leaderboard):
        cls._write_json(LEADERBOARD_FILE, leaderboard)

    @classmethod
    def load_promotions(cls):
        return cls._read_json(PROMOTION_FILE, [])

    @classmethod
    def save_promotions(cls, promotions):
        cls._write_json(PROMOTION_FILE, promotions)


class AttendanceView(discord.ui.View):
    def __init__(self, cog: "AttendanceSystem", meet_id: str):
        super().__init__(timeout=None)
        self.cog = cog
        self.meet_id = meet_id

    async def handle_vote(self, interaction: discord.Interaction, status: str):
        if not interaction.user or interaction.user.bot:
            return

        meet = self.cog.meets.get(self.meet_id)
        if not meet:
            await interaction.response.send_message("This meet record no longer exists.", ephemeral=True)
            return

        if meet.closed:
            await interaction.response.send_message("This attendance panel is already closed.", ephemeral=True)
            return

        user_id = interaction.user.id
        meet.attendees_yes.discard(user_id)
        meet.attendees_maybe.discard(user_id)
        meet.attendees_no.discard(user_id)

        if status == "yes":
            meet.attendees_yes.add(user_id)
            msg = "You are marked as **Pulling Up** ✅"
        elif status == "maybe":
            meet.attendees_maybe.add(user_id)
            msg = "You are marked as **Maybe** ❓"
        else:
            meet.attendees_no.add(user_id)
            msg = "You are marked as **Can't Make It** ❌"

        self.cog.save_all()
        await self.cog.refresh_meet_message(meet)
        await interaction.response.send_message(msg, ephemeral=True)

    @discord.ui.button(label="Pulling Up", emoji="✅", style=discord.ButtonStyle.success, custom_id="attendance_yes")
    async def pulling_up(self, interaction: discord.Interaction, button: discord.ui.Button):
        await self.handle_vote(interaction, "yes")

    @discord.ui.button(label="Maybe", emoji="❓", style=discord.ButtonStyle.secondary, custom_id="attendance_maybe")
    async def maybe(self, interaction: discord.Interaction, button: discord.ui.Button):
        await self.handle_vote(interaction, "maybe")

    @discord.ui.button(label="Can't Make It", emoji="❌", style=discord.ButtonStyle.danger, custom_id="attendance_no")
    async def cant_make_it(self, interaction: discord.Interaction, button: discord.ui.Button):
        await self.handle_vote(interaction, "no")


class AttendanceSystem(commands.Cog):
    def __init__(self, bot: commands.Bot):
        self.bot = bot
        self.meets: Dict[str, MeetRecord] = AttendanceStorage.load_meets()
        self.leaderboard = AttendanceStorage.load_leaderboard()
        self.promotions = AttendanceStorage.load_promotions()

        # Re-register persistent views for active panels after restart.
        for meet_id, meet in self.meets.items():
            if not meet.closed:
                self.bot.add_view(AttendanceView(self, meet_id))

    def save_all(self):
        AttendanceStorage.save_meets(self.meets)
        AttendanceStorage.save_leaderboard(self.leaderboard)
        AttendanceStorage.save_promotions(self.promotions)

    def _make_meet_id(self):
        return datetime.now(timezone.utc).strftime("meet_%Y%m%d_%H%M%S")

    def _build_embed(self, meet: MeetRecord) -> discord.Embed:
        embed = discord.Embed(
            title="📊 DIFF Meet Attendance",
            description=(
                "Use the buttons below to update your status for this meet.\n\n"
                "━━━━━━━━━━━━━━━━━━━━━━"
            ),
            color=discord.Color.blue(),
            timestamp=datetime.now(timezone.utc),
        )
        embed.add_field(name="Meet", value=meet.title, inline=True)
        embed.add_field(name="Host", value=meet.host_name, inline=True)
        embed.add_field(name="Date", value=meet.meet_date, inline=True)
        embed.add_field(name="✅ Pulling Up", value=str(len(meet.attendees_yes)), inline=True)
        embed.add_field(name="❓ Maybe", value=str(len(meet.attendees_maybe)), inline=True)
        embed.add_field(name="❌ Can't Make It", value=str(len(meet.attendees_no)), inline=True)
        embed.add_field(name="✅ Checked In", value=str(len(meet.checked_in)), inline=True)
        embed.add_field(name="Status", value="Closed 🔒" if meet.closed else "Open ✅", inline=True)
        embed.add_field(name="Meet ID", value=meet.meet_id, inline=True)
        embed.set_footer(text="Different Meets • Attendance System")
        return embed

    async def refresh_meet_message(self, meet: MeetRecord):
        channel = self.bot.get_channel(meet.channel_id)
        if not isinstance(channel, discord.TextChannel):
            return
        if not meet.message_id:
            return
        try:
            message = await channel.fetch_message(meet.message_id)
            view = None if meet.closed else AttendanceView(self, meet.meet_id)
            await message.edit(embed=self._build_embed(meet), view=view)
        except discord.NotFound:
            pass
        except discord.HTTPException:
            pass

    async def update_member_stats(self, guild: discord.Guild, user_id: int):
        member = guild.get_member(user_id)
        if not member:
            return
        entry = self.leaderboard.get(str(user_id), {
            "user_id": user_id,
            "name": member.display_name,
            "attendance_count": 0,
            "current_role": self.get_top_tracked_role(member),
            "last_attended": None,
        })
        entry["name"] = member.display_name
        entry["attendance_count"] += 1
        entry["current_role"] = self.get_top_tracked_role(member)
        entry["last_attended"] = datetime.now(timezone.utc).isoformat()
        self.leaderboard[str(user_id)] = entry
        self.check_promotion(member, entry)
        self.save_all()

    def get_top_tracked_role(self, member: discord.Member) -> str:
        priority = ["Leader", "Co-Leader", "Manager", "Host", "Crew Member"]
        member_role_names = {role.name for role in member.roles}
        for role_name in priority:
            if role_name in member_role_names:
                return role_name
        return "Crew Member"

    def check_promotion(self, member: discord.Member, entry: dict):
        current_role = entry.get("current_role", "Crew Member")
        threshold = PROMOTION_THRESHOLDS.get(current_role)
        next_role = PROMOTION_PATH.get(current_role)
        if not threshold or not next_role:
            return
        if entry["attendance_count"] < threshold:
            return

        already_logged = any(
            p.get("user_id") == member.id and p.get("suggested_role") == next_role
            for p in self.promotions
        )
        if already_logged:
            return

        self.promotions.append({
            "user_id": member.id,
            "name": member.display_name,
            "current_role": current_role,
            "suggested_role": next_role,
            "attendance_count": entry["attendance_count"],
            "created_at": datetime.now(timezone.utc).isoformat(),
        })

    @app_commands.command(name="attendance_create", description="Create a new DIFF attendance panel.")
    @app_commands.describe(meet_title="Meet name", host="Host for the meet", meet_date="Date shown on the panel")
    async def attendance_create(self, interaction: discord.Interaction, meet_title: str, host: discord.Member, meet_date: str):
        channel = interaction.client.get_channel(ATTENDANCE_CHANNEL_ID)
        if not isinstance(channel, discord.TextChannel):
            await interaction.response.send_message(
                f"Attendance channel not found. Update ATTENDANCE_CHANNEL_ID in the file.",
                ephemeral=True,
            )
            return

        meet_id = self._make_meet_id()
        meet = MeetRecord(
            meet_id=meet_id,
            title=meet_title,
            host_id=host.id,
            host_name=host.mention,
            meet_date=meet_date,
            created_at=datetime.now(timezone.utc).isoformat(),
            channel_id=channel.id,
        )
        self.meets[meet_id] = meet

        view = AttendanceView(self, meet_id)
        message = await channel.send(embed=self._build_embed(meet), view=view)
        meet.message_id = message.id
        self.bot.add_view(AttendanceView(self, meet_id))
        self.save_all()

        await interaction.response.send_message(
            f"Attendance panel created in {channel.mention} for **{meet_title}**.",
            ephemeral=True,
        )

    @app_commands.command(name="attendance_checkin", description="Mark a member as actually present after the meet starts.")
    @app_commands.describe(meet_id="Meet ID from the attendance panel", member="Member to check in")
    async def attendance_checkin(self, interaction: discord.Interaction, meet_id: str, member: discord.Member):
        meet = self.meets.get(meet_id)
        if not meet:
            await interaction.response.send_message("Meet ID not found.", ephemeral=True)
            return

        meet.checked_in.add(member.id)
        self.save_all()
        await self.update_member_stats(interaction.guild, member.id)
        await self.refresh_meet_message(meet)
        await interaction.response.send_message(
            f"Checked in {member.mention} for **{meet.title}**.",
            ephemeral=True,
        )

    @app_commands.command(name="attendance_close", description="Close attendance and post final results.")
    @app_commands.describe(meet_id="Meet ID from the attendance panel", total_players="Total players in lobby")
    async def attendance_close(self, interaction: discord.Interaction, meet_id: str, total_players: int):
        meet = self.meets.get(meet_id)
        if not meet:
            await interaction.response.send_message("Meet ID not found.", ephemeral=True)
            return

        meet.closed = True
        self.save_all()
        await self.refresh_meet_message(meet)

        channel = interaction.client.get_channel(meet.channel_id)
        if isinstance(channel, discord.TextChannel):
            result_embed = discord.Embed(
                title="📊 DIFF Meet Results",
                description="Final attendance results for this meet.",
                color=discord.Color.green(),
                timestamp=datetime.now(timezone.utc),
            )
            result_embed.add_field(name="Meet", value=meet.title, inline=True)
            result_embed.add_field(name="Host", value=meet.host_name, inline=True)
            result_embed.add_field(name="Lobby Size", value=str(total_players), inline=True)
            result_embed.add_field(name="✅ Checked In", value=str(len(meet.checked_in)), inline=True)
            result_embed.add_field(name="❌ No Shows", value=str(max(0, len(meet.attendees_yes) - len(meet.checked_in))), inline=True)
            result_embed.add_field(name="❓ Maybe", value=str(len(meet.attendees_maybe)), inline=True)
            result_embed.set_footer(text="Attach your lobby screenshot below this post.")
            await channel.send(embed=result_embed)

        await interaction.response.send_message(
            f"Attendance closed for **{meet.title}** and final results were posted.",
            ephemeral=True,
        )

    @app_commands.command(name="attendance_leaderboard", description="Show the most active DIFF members by attendance.")
    async def attendance_leaderboard(self, interaction: discord.Interaction):
        if not self.leaderboard:
            await interaction.response.send_message("No attendance data yet.", ephemeral=True)
            return

        top_entries = sorted(
            self.leaderboard.values(),
            key=lambda x: x.get("attendance_count", 0),
            reverse=True,
        )[:10]

        lines = []
        medals = ["🥇", "🥈", "🥉"]
        for idx, entry in enumerate(top_entries, start=1):
            prefix = medals[idx - 1] if idx <= 3 else f"{idx}."
            lines.append(
                f"{prefix} <@{entry['user_id']}> — **{entry['attendance_count']}** meet(s) attended"
            )

        embed = discord.Embed(
            title="🏆 DIFF Attendance Leaderboard",
            description="\n".join(lines),
            color=discord.Color.gold(),
            timestamp=datetime.now(timezone.utc),
        )
        embed.set_footer(text="Attendance is tracked from check-ins, not button votes.")
        await interaction.response.send_message(embed=embed)

    @app_commands.command(name="attendance_promotions", description="Show promotion suggestions based on attendance.")
    async def attendance_promotions(self, interaction: discord.Interaction):
        if not self.promotions:
            await interaction.response.send_message("No promotion suggestions yet.", ephemeral=True)
            return

        lines = []
        for item in self.promotions[-10:][::-1]:
            lines.append(
                f"📈 <@{item['user_id']}> — {item['current_role']} → **{item['suggested_role']}** "
                f"({item['attendance_count']} attends)"
            )

        embed = discord.Embed(
            title="📈 Promotion Suggestions",
            description="\n".join(lines),
            color=discord.Color.purple(),
            timestamp=datetime.now(timezone.utc),
        )
        embed.set_footer(text="Review suggestions manually before promoting anyone.")
        await interaction.response.send_message(embed=embed)


async def setup(bot: commands.Bot):
    await bot.add_cog(AttendanceSystem(bot))
