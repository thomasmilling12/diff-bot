"""
DIFF Host Flow System — Smart Ping Version
Drop-in discord.py 2.x module.

What changed:
- Smart ping system for PS5 + Car Meet roles
- Role ping is sent on Start Meet only
- End Meet does NOT ping by default
- Prevents duplicate role pings within a cooldown window
- Uses host Discord mention instead of PSN text
- Includes Host Hub panel with:
    🔱 Start Meet
    📌 End Meet
    🎤 Voice Script

SETUP:
1) Replace the role IDs below with your real Discord role IDs.
2) Put this file next to your current bot file.
3) Load it as an extension, or call setup(bot).
4) Run !send_host_hub to post the panel.

If you already use persistent views, make sure bot.add_view(HostHubView(bot)) is called on startup.
"""

from __future__ import annotations

import asyncio
from datetime import datetime
from typing import Optional

import discord
from discord.ext import commands

# =========================
# CONFIG
# =========================

MEET_FLOW_CHANNEL_ID = 1485684577073758378

# Optional private logs channel. If None, logs go to the meet flow channel.
LOG_CHANNEL_ID: Optional[int] = None

# Replace these with your real role IDs
PS5_ROLE_ID = 0
CAR_MEET_ROLE_ID = 0

# Who can use the Host Hub buttons. Leave empty to allow anyone.
ALLOWED_ROLE_IDS: list[int] = []

# Cooldown for button presses by the same user
BUTTON_COOLDOWN_SECONDS = 5

# Smart ping settings
PING_ON_START_MEET = True
PING_ON_END_MEET = False
SMART_PING_COOLDOWN_MINUTES = 60  # prevents repeat role pings too soon


# =========================
# HELPERS
# =========================

def has_allowed_role(member: discord.Member) -> bool:
    if not ALLOWED_ROLE_IDS:
        return True
    member_role_ids = {role.id for role in member.roles}
    return any(role_id in member_role_ids for role_id in ALLOWED_ROLE_IDS)


def get_timestamp_strings() -> tuple[str, str]:
    now = datetime.now()
    return now.strftime("%b %d, %Y"), now.strftime("%I:%M %p")


def get_role_mentions(guild: discord.Guild) -> str:
    mentions: list[str] = []

    ps5_role = guild.get_role(PS5_ROLE_ID) if PS5_ROLE_ID else None
    car_meet_role = guild.get_role(CAR_MEET_ROLE_ID) if CAR_MEET_ROLE_ID else None

    if ps5_role:
        mentions.append(ps5_role.mention)
    if car_meet_role:
        mentions.append(car_meet_role.mention)

    return " ".join(mentions).strip()

def build_host_log_message(host_mention: str, ping_used: bool) -> str:
    date_str, time_str = get_timestamp_strings()
    ping_status = "Yes" if ping_used else "No"
    return (
        "📊 __**DIFF Host Log**__ 📊\n\n"
        f"**Host:** {host_mention}\n"
        "**Flow Completed:** Meet Ended\n"
        f"**Role Ping Sent:** {ping_status}\n"
        f"**Logged At:** {time_str} | {date_str}\n\n"
        "— **Different Meets**"
    )

class HostHubView(discord.ui.View):
    def __init__(self, bot: commands.Bot):
        super().__init__(timeout=None)
        self.bot = bot
        self._user_button_cooldowns: dict[int, float] = {}
        self._guild_role_ping_times: dict[int, float] = {}

    async def interaction_check(self, interaction: discord.Interaction) -> bool:
        if not interaction.guild or not isinstance(interaction.user, discord.Member):
            await interaction.response.send_message(
                "This panel can only be used inside the server.",
                ephemeral=True,
            )
            return False

        if not has_allowed_role(interaction.user):
            await interaction.response.send_message(
                "You do not have permission to use this host panel.",
                ephemeral=True,
            )
            return False

        now = asyncio.get_event_loop().time()
        last_used = self._user_button_cooldowns.get(interaction.user.id, 0)
        if now - last_used < BUTTON_COOLDOWN_SECONDS:
            remaining = round(BUTTON_COOLDOWN_SECONDS - (now - last_used), 1)
            await interaction.response.send_message(
                f"Please wait {remaining} seconds before pressing another host button.",
                ephemeral=True,
            )
            return False

        self._user_button_cooldowns[interaction.user.id] = now
        return True

    async def _get_target_channel(self, guild: discord.Guild, channel_id: int) -> Optional[discord.TextChannel]:
        channel = guild.get_channel(channel_id)
        if channel is None:
            fetched = self.bot.get_channel(channel_id)
            if isinstance(fetched, discord.TextChannel):
                channel = fetched
        return channel if isinstance(channel, discord.TextChannel) else None

    def _should_send_role_ping(self, guild_id: int) -> bool:
        now = asyncio.get_event_loop().time()
        last_ping = self._guild_role_ping_times.get(guild_id, 0)
        cooldown_seconds = SMART_PING_COOLDOWN_MINUTES * 60
        if now - last_ping >= cooldown_seconds:
            self._guild_role_ping_times[guild_id] = now
            return True
        return False

    @discord.ui.button(
        label="Start Meet",
        emoji="🔱",
        style=discord.ButtonStyle.success,
        custom_id="diff_hosthub_start_meet_smart",
    )
    async def start_meet_button(self, interaction: discord.Interaction, button: discord.ui.Button):
        assert interaction.guild is not None
        assert isinstance(interaction.user, discord.Member)

        target_channel = await self._get_target_channel(interaction.guild, MEET_FLOW_CHANNEL_ID)
        if target_channel is None:
            await interaction.response.send_message(
                "I could not find the meet flow channel. Check MEET_FLOW_CHANNEL_ID in the file.",
                ephemeral=True,
            )
            return

        ping_text = ""
        ping_sent = False

        if PING_ON_START_MEET:
            if self._should_send_role_ping(interaction.guild.id):
                ping_text = get_role_mentions(interaction.guild)
                ping_sent = bool(ping_text)

        message = build_start_meet_message(interaction.user.mention, ping_text)
        await target_channel.send(
            message,
            allowed_mentions=discord.AllowedMentions(roles=True, users=False, everyone=False),
        )

        extra = " Smart ping sent." if ping_sent else " No role ping was sent."
        await interaction.response.send_message(
            f"Start Meet flow sent in {target_channel.mention}.{extra}",
            ephemeral=True,
        )

    @discord.ui.button(
        label="End Meet",
        emoji="📌",
        style=discord.ButtonStyle.danger,
        custom_id="diff_hosthub_end_meet_smart",
    )
    async def end_meet_button(self, interaction: discord.Interaction, button: discord.ui.Button):
        assert interaction.guild is not None
        assert isinstance(interaction.user, discord.Member)

        target_channel = await self._get_target_channel(interaction.guild, MEET_FLOW_CHANNEL_ID)
        if target_channel is None:
            await interaction.response.send_message(
                "I could not find the meet flow channel. Check MEET_FLOW_CHANNEL_ID in the file.",
                ephemeral=True,
            )
            return

        ping_text = ""
        ping_sent = False

        if PING_ON_END_MEET:
            if self._should_send_role_ping(interaction.guild.id):
                ping_text = get_role_mentions(interaction.guild)
                ping_sent = bool(ping_text)

        await target_channel.send(
            build_end_meet_message(ping_text),
            allowed_mentions=discord.AllowedMentions(roles=True, users=False, everyone=False),
        )

        log_channel_id = LOG_CHANNEL_ID or MEET_FLOW_CHANNEL_ID
        log_channel = await self._get_target_channel(interaction.guild, log_channel_id)
        if log_channel is not None:
            await log_channel.send(build_host_log_message(interaction.user.mention, ping_sent))

        extra = " Smart ping sent." if ping_sent else " No role ping was sent."
        await interaction.response.send_message(
            f"End Meet flow sent in {target_channel.mention} and host activity was logged.{extra}",
            ephemeral=True,
        )

    @discord.ui.button(
        label="Voice Script",
        emoji="🎤",
        style=discord.ButtonStyle.primary,
        custom_id="diff_hosthub_voice_script_smart",
    )
    async def voice_script_button(self, interaction: discord.Interaction, button: discord.ui.Button):
        assert isinstance(interaction.user, discord.Member)
        await interaction.response.send_message(
            build_voice_script_message(interaction.user.mention),
            ephemeral=True,
        )


# =========================
# COG
# =========================

class HostHubCog(commands.Cog):
    def __init__(self, bot: commands.Bot):
        self.bot = bot
        self.bot.add_view(HostHubView(bot))

    @commands.command(name="send_host_hub")
    @commands.guild_only()
    async def send_host_hub(self, ctx: commands.Context):
        if not isinstance(ctx.author, discord.Member):
            return

        if not has_allowed_role(ctx.author):
            await ctx.reply("You do not have permission to send the Host Hub panel.")
            return

        embed = discord.Embed(
            title="📌 DIFF Host Flow System",
            description=(
                "*Use the buttons below to manage your meet flow professionally.*\n\n"
                "🔱 **Start Meet** — Sends the official DIFF welcome speech\n"
                "📌 **End Meet** — Sends the official ending speech and logs your host activity\n"
                "🎤 **Voice Script** — Displays the short host script for easier speaking during meets\n\n"
                "━━━━━━━━━━━━━━━━━━━━━━\n\n"
                f"**Smart Ping:** Role ping on start = {'ON' if PING_ON_START_MEET else 'OFF'} | "
                f"Role ping on end = {'ON' if PING_ON_END_MEET else 'OFF'} | "
                f"Cooldown = {SMART_PING_COOLDOWN_MINUTES} min\n\n"
                "*Keep hosting clean, organized, and professional.*\n\n"
                "— **Different Meets**"
            ),
            color=discord.Color.blurple(),
        )
        embed.set_footer(text="DIFF Host Hub")

        await ctx.send(embed=embed, view=HostHubView(self.bot))

    @commands.command(name="host_voice_script")
    @commands.guild_only()
    async def host_voice_script(self, ctx: commands.Context):
        if not isinstance(ctx.author, discord.Member):
            return

        if not has_allowed_role(ctx.author):
            await ctx.reply("You do not have permission to use this command.")
            return

        await ctx.send(build_voice_script_message(ctx.author.mention))


async def setup(bot: commands.Bot):
    await bot.add_cog(HostHubCog(bot))
