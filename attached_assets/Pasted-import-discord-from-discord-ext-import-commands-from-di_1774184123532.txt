# =========================================================
# STORAGE
# =========================================================
def load_data():
    if not os.path.exists(APPLICATIONS_FILE):
        return {"last_id": 0, "applications": {}}
    with open(APPLICATIONS_FILE, "r", encoding="utf-8") as f:
        return json.load(f)


def save_data(data):
    with open(APPLICATIONS_FILE, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=2)


def create_next_id():
    data = load_data()
    data["last_id"] += 1
    new_id = str(data["last_id"]).zfill(4)
    save_data(data)
    return new_id


def get_app(app_id: str) -> Optional[dict]:
    data = load_data()
    return data["applications"].get(app_id)


def save_app(app_id: str, payload: dict):
    data = load_data()
    data["applications"][app_id] = payload
    save_data(data)


def update_app(app_id: str, **updates):
    data = load_data()
    if app_id in data["applications"]:
        data["applications"][app_id].update(updates)
        save_data(data)

def is_staff_reviewer(member: discord.Member) -> bool:
    allowed = {LEADER_ROLE_ID, CO_LEADER_ROLE_ID, MANAGER_ROLE_ID}
    return any(role.id in allowed for role in member.roles)


def get_staff_role_ids():
    return [LEADER_ROLE_ID, CO_LEADER_ROLE_ID, MANAGER_ROLE_ID]


def utc_now():
    return datetime.utcnow()


def make_status_emoji(status: str) -> str:
    mapping = {
        "Pending": "🟡 Pending",
        "Approved": "🟢 Approved",
        "Denied": "🔴 Denied",
        "Closed": "⚫ Closed",
    }
    return mapping.get(status, status)


async def safe_dm(user: discord.User | discord.Member, message: str):
    try:
        await user.send(message)
    except Exception:
        pass

def build_review_embed(app_id: str, applicant: discord.Member | discord.User, answers: dict, ticket_channel_id: Optional[int] = None):
    embed = discord.Embed(
        title=f"DIFF Application #{app_id}",
        description="Staff review the application and check the applicant garage ticket before making a decision.",
        color=discord.Color.blurple(),
        timestamp=utc_now(),
    )
    embed.add_field(name="Applicant", value=f"{applicant.mention}\n`{applicant.id}`", inline=False)
    embed.add_field(name="Gamertag / Username", value=answers["gamer_tag"], inline=True)
    embed.add_field(name="Age", value=answers["age"], inline=True)
    embed.add_field(name="Discord", value=str(applicant), inline=True)
    embed.add_field(name="Build Style", value=answers["car_style"], inline=False)
    embed.add_field(name="Discord Activity", value=answers["discord_activity"], inline=False)
    embed.add_field(name="Why Join DIFF", value=answers["reason"], inline=False)
    if ticket_channel_id:
        embed.add_field(name="Garage Ticket", value=f"<#{ticket_channel_id}>", inline=False)
    embed.set_footer(text="Status: Pending Review")
    return embed


def build_tracker_embed(app_id: str, applicant: discord.Member | discord.User, answers: dict, status: str, reviewer_text: str = "Not reviewed yet"):
    color = discord.Color.orange()
    if status == "Approved":
        color = discord.Color.green()
    elif status == "Denied":
        color = discord.Color.red()
    elif status == "Closed":
        color = discord.Color.dark_grey()

    embed = discord.Embed(
        title=f"Application Tracker #{app_id}",
        description="DIFF application progress",
        color=color,
        timestamp=utc_now(),
    )
    embed.add_field(name="User", value=applicant.mention, inline=True)
    embed.add_field(name="Gamertag", value=answers["gamer_tag"], inline=True)
    embed.add_field(name="Status", value=make_status_emoji(status), inline=True)
    embed.add_field(name="Reviewed By", value=reviewer_text, inline=False)
    embed.set_footer(text=f"Applicant ID: {applicant.id}")
    return embed

class ApplyPanelView(discord.ui.View):
    def __init__(self, cog: "DiffApplicationSystem"):
        super().__init__(timeout=None)
        self.cog = cog

    @discord.ui.button(label="Apply to DIFF", style=discord.ButtonStyle.primary, custom_id="diff_apply_open_modal")
    async def apply_button(self, interaction: discord.Interaction, button: discord.ui.Button):
        await interaction.response.send_modal(ApplicationModal(self.cog))


class ReviewView(discord.ui.View):
    def __init__(self, cog: "DiffApplicationSystem", app_id: str, applicant_id: int):
        super().__init__(timeout=None)
        self.cog = cog
        self.app_id = app_id
        self.applicant_id = applicant_id

    @discord.ui.button(label="Accept", style=discord.ButtonStyle.success, custom_id="diff_review_accept")
    async def accept_button(self, interaction: discord.Interaction, button: discord.ui.Button):
        if not interaction.guild or not isinstance(interaction.user, discord.Member):
            return await interaction.response.send_message("This action only works inside the server.", ephemeral=True)

        if not is_staff_reviewer(interaction.user):
            return await interaction.response.send_message(
                "Only Leader, Co-Leader, or Manager can approve applications.",
                ephemeral=True,
            )

        record = get_app(self.app_id)
        if not record:
            return await interaction.response.send_message("Application record not found.", ephemeral=True)

        if record.get("status") != "Pending":
            return await interaction.response.send_message("This application has already been reviewed.", ephemeral=True)

        applicant = interaction.guild.get_member(self.applicant_id)
        if applicant is None:
            return await interaction.response.send_message("Applicant is no longer in the server.", ephemeral=True)

        approved_role = interaction.guild.get_role(APPROVED_MEMBER_ROLE_ID)
        if approved_role is None:
            return await interaction.response.send_message("Approved member role is not configured correctly.", ephemeral=True)

        try:
            await applicant.add_roles(approved_role, reason=f"DIFF application #{self.app_id} approved by {interaction.user}")
        except discord.Forbidden:
            return await interaction.response.send_message("I do not have permission to assign that role.", ephemeral=True)

        await self.finalize_review(interaction, "Approved", interaction.user, close_ticket=True)
        await safe_dm(applicant, f"Your DIFF application **#{self.app_id}** was **approved**. Welcome to DIFF. 🎉")

    @discord.ui.button(label="Deny", style=discord.ButtonStyle.danger, custom_id="diff_review_deny")
    async def deny_button(self, interaction: discord.Interaction, button: discord.ui.Button):
        if not interaction.guild or not isinstance(interaction.user, discord.Member):
            return await interaction.response.send_message("This action only works inside the server.", ephemeral=True)

        if not is_staff_reviewer(interaction.user):
            return await interaction.response.send_message(
                "Only Leader, Co-Leader, or Manager can deny applications.",
                ephemeral=True,
            )

        record = get_app(self.app_id)
        if not record:
            return await interaction.response.send_message("Application record not found.", ephemeral=True)

        if record.get("status") != "Pending":
            return await interaction.response.send_message("This application has already been reviewed.", ephemeral=True)

        applicant = interaction.guild.get_member(self.applicant_id)
        await self.finalize_review(interaction, "Denied", interaction.user, close_ticket=True)
        if applicant:
            await safe_dm(applicant, f"Your DIFF application **#{self.app_id}** was **denied**.")

    async def finalize_review(self, interaction: discord.Interaction, new_status: str, reviewer: discord.Member, close_ticket: bool):
        for child in self.children:
            child.disabled = True

        record = get_app(self.app_id)
        if not record:
            return await interaction.response.send_message("Application record not found.", ephemeral=True)

        update_app(
            self.app_id,
            status=new_status,
            reviewed_by=str(reviewer),
            reviewed_by_id=reviewer.id,
            reviewed_at=utc_now().isoformat(),
        )

        embed = interaction.message.embeds[0]
        if new_status == "Approved":
            embed.color = discord.Color.green()
        elif new_status == "Denied":
            embed.color = discord.Color.red()
        embed.set_footer(text=f"Status: {new_status} • Reviewed by {reviewer}")
        embed.timestamp = utc_now()

        await interaction.response.edit_message(embed=embed, view=self)

        # Update tracker
        guild = interaction.guild
        if guild:
            tracker_channel = guild.get_channel(APPLICATION_TRACKER_CHANNEL_ID)
            if tracker_channel and record.get("tracker_message_id"):
                try:
                    tracker_msg = await tracker_channel.fetch_message(record["tracker_message_id"])
                    applicant = guild.get_member(record["user_id"]) or await guild.fetch_member(record["user_id"])
                    tracker_embed = build_tracker_embed(
                        self.app_id,
                        applicant,
                        {
                            "gamer_tag": record["gamer_tag"],
                        },
                        new_status,
                        reviewer.mention,
                    )
                    await tracker_msg.edit(embed=tracker_embed)
                except Exception:
                    pass

        # Close ticket
        if close_ticket and guild and record.get("ticket_channel_id"):
            channel = guild.get_channel(record["ticket_channel_id"])
            if isinstance(channel, discord.TextChannel):
                try:
                    await channel.send(
                        embed=discord.Embed(
                            title="Application Decision Complete",
                            description=f"This application has been **{new_status.lower()}**.\nThis ticket will now be closed.",
                            color=discord.Color.green() if new_status == "Approved" else discord.Color.red(),
                            timestamp=utc_now(),
                        )
                    )
                except Exception:
                    pass

                try:
                    await channel.edit(name=f"closed-{channel.name[:80]}")
                except Exception:
                    pass

                try:
                    await channel.set_permissions(guild.default_role, view_channel=False)
                except Exception:
                    pass

                # Remove applicant access after decision
                try:
                    applicant = guild.get_member(record["user_id"])
                    if applicant:
                        await channel.set_permissions(applicant, overwrite=None)
                except Exception:
                    pass

                update_app(self.app_id, ticket_closed=True, closed_at=utc_now().isoformat())

class ApplicationModal(discord.ui.Modal, title="DIFF Crew Application"):
    gamer_tag = discord.ui.TextInput(
        label="Gamertag / Username",
        placeholder="Enter your PSN / username",
        max_length=100,
        required=True,
    )

    age = discord.ui.TextInput(
        label="Age",
        placeholder="Must be 18+",
        max_length=3,
        required=True,
    )

    car_style = discord.ui.TextInput(
        label="What type of builds do you make?",
        placeholder="Clean, realistic, JDM, stance, drift, etc.",
        style=discord.TextStyle.paragraph,
        max_length=500,
        required=True,
    )

    discord_activity = discord.ui.TextInput(
        label="How active are you on Discord?",
        placeholder="Tell us how often you check Discord",
        style=discord.TextStyle.paragraph,
        max_length=500,
        required=True,
    )

    reason = discord.ui.TextInput(
        label="Why do you want to join DIFF?",
        placeholder="Tell staff why you want to join",
        style=discord.TextStyle.paragraph,
        max_length=1000,
        required=True,
    )

    def __init__(self, cog: "DiffApplicationSystem"):
        super().__init__()
        self.cog = cog

    async def on_submit(self, interaction: discord.Interaction):
        guild = interaction.guild
        if guild is None:
            return await interaction.response.send_message("This can only be used inside the DIFF server.", ephemeral=True)

        # Basic age validation
        try:
            age_num = int(str(self.age))
            if age_num < 18:
                return await interaction.response.send_message("DIFF applications are only for members who are 18+.", ephemeral=True)
        except ValueError:
            return await interaction.response.send_message("Please enter a valid age number.", ephemeral=True)

        review_channel = guild.get_channel(APPLICATION_REVIEW_CHANNEL_ID)
        tracker_channel = guild.get_channel(APPLICATION_TRACKER_CHANNEL_ID)
        category = guild.get_channel(APPLICATION_TICKET_CATEGORY_ID)

        if not isinstance(review_channel, discord.TextChannel):
            return await interaction.response.send_message("Application review channel is not configured correctly.", ephemeral=True)

        if not isinstance(tracker_channel, discord.TextChannel):
            return await interaction.response.send_message("Application tracker channel is not configured correctly.", ephemeral=True)

        if APPLICATION_TICKET_CATEGORY_ID == 0 or not isinstance(category, discord.CategoryChannel):
            return await interaction.response.send_message(
                "Ticket category is not configured yet. Set APPLICATION_TICKET_CATEGORY_ID first.",
                ephemeral=True,
            )

        app_id = create_next_id()
        answers = {
            "gamer_tag": str(self.gamer_tag),
            "age": str(self.age),
            "car_style": str(self.car_style),
            "discord_activity": str(self.discord_activity),
            "reason": str(self.reason),
        }

        # Create private garage ticket
        overwrites = {
            guild.default_role: discord.PermissionOverwrite(view_channel=False),
            interaction.user: discord.PermissionOverwrite(view_channel=True, send_messages=True, attach_files=True, read_message_history=True),
            guild.me: discord.PermissionOverwrite(view_channel=True, send_messages=True, manage_channels=True, read_message_history=True),
        }

        for role_id in get_staff_role_ids():
            role = guild.get_role(role_id)
            if role:
                overwrites[role] = discord.PermissionOverwrite(view_channel=True, send_messages=True, read_message_history=True)

        ticket_name = f"garage-{app_id}-{interaction.user.name}".lower().replace(" ", "-")[:95]

        try:
            ticket_channel = await guild.create_text_channel(
                name=ticket_name,
                category=category,
                overwrites=overwrites,
                topic=f"DIFF Garage Ticket for Application #{app_id} | User ID: {interaction.user.id}",
                reason=f"Garage ticket created for DIFF application #{app_id}",
            )
        except discord.Forbidden:
            return await interaction.response.send_message("I do not have permission to create ticket channels.", ephemeral=True)

        # Send ticket intro
        ticket_embed = discord.Embed(
            title=f"Garage Submission Ticket #{app_id}",
            description=(
                f"Welcome {interaction.user.mention}.\n\n"
                "**Next step:** Upload clear pictures of your garage / cars in this ticket.\n\n"
                "Staff will review:\n"
                "• Your application answers\n"
                "• Your overall garage quality\n"
                "• Build realism / cleanliness\n\n"
                "Once staff finishes reviewing, this ticket will be closed automatically."
            ),
            color=discord.Color.blue(),
            timestamp=utc_now(),
        )
        await ticket_channel.send(content=f"{interaction.user.mention}", embed=ticket_embed)

        # Send review message
        review_embed = build_review_embed(app_id, interaction.user, answers, ticket_channel.id)
        review_view = ReviewView(self.cog, app_id=app_id, applicant_id=interaction.user.id)
        review_message = await review_channel.send(embed=review_embed, view=review_view)

        # Send tracker message
        tracker_embed = build_tracker_embed(app_id, interaction.user, {"gamer_tag": answers["gamer_tag"]}, "Pending")
        tracker_message = await tracker_channel.send(embed=tracker_embed)

        # Save record
        save_app(app_id, {
            "app_id": app_id,
            "user_id": interaction.user.id,
            "username": str(interaction.user),
            "gamer_tag": answers["gamer_tag"],
            "age": answers["age"],
            "car_style": answers["car_style"],
            "discord_activity": answers["discord_activity"],
            "reason": answers["reason"],
            "status": "Pending",
            "submitted_at": utc_now().isoformat(),
            "review_channel_id": review_channel.id,
            "review_message_id": review_message.id,
            "tracker_channel_id": tracker_channel.id,
            "tracker_message_id": tracker_message.id,
            "ticket_channel_id": ticket_channel.id,
            "ticket_channel_name": ticket_channel.name,
            "reviewed_by": None,
            "reviewed_at": None,
            "ticket_closed": False,
        })

        await interaction.response.send_message(
            f"Your DIFF application has been submitted.\n"
            f"Application ID: **#{app_id}**\n"
            f"Garage Ticket: {ticket_channel.mention}\n\n"
            f"Please upload your garage pics in that ticket.",
            ephemeral=True,
        )

class DiffApplicationSystem(commands.Cog):
    def __init__(self, bot: commands.Bot):
        self.bot = bot

    async def cog_load(self):
        self.bot.add_view(ApplyPanelView(self))

    @app_commands.command(name="apply", description="Open the DIFF application form")
    async def apply(self, interaction: discord.Interaction):
        await interaction.response.send_modal(ApplicationModal(self))

    @app_commands.command(name="post_apply_panel", description="Post the DIFF apply panel")
    async def post_apply_panel(self, interaction: discord.Interaction):
        if not isinstance(interaction.user, discord.Member) or not is_staff_reviewer(interaction.user):
            return await interaction.response.send_message(
                "Only Leader, Co-Leader, or Manager can use this command.",
                ephemeral=True,
            )

        channel = interaction.channel
        if APPLICATION_PANEL_CHANNEL_ID != 0:
            forced_channel = interaction.guild.get_channel(APPLICATION_PANEL_CHANNEL_ID) if interaction.guild else None
            if isinstance(forced_channel, discord.TextChannel):
                channel = forced_channel

        if not isinstance(channel, discord.TextChannel):
            return await interaction.response.send_message("Could not find a valid channel to post the panel.", ephemeral=True)

        embed = build_panel_embed()
        await channel.send(embed=embed, view=ApplyPanelView(self))
        await interaction.response.send_message(f"DIFF application panel posted in {channel.mention}.", ephemeral=True)

    @app_commands.command(name="application_lookup", description="Look up a DIFF application by ID")
    @app_commands.describe(application_id="Example: 0001")
    async def application_lookup(self, interaction: discord.Interaction, application_id: str):
        if not isinstance(interaction.user, discord.Member) or not is_staff_reviewer(interaction.user):
            return await interaction.response.send_message(
                "Only Leader, Co-Leader, or Manager can use this command.",
                ephemeral=True,
            )

        record = get_app(application_id)
        if not record:
            return await interaction.response.send_message(f"No application found with ID #{application_id}.", ephemeral=True)

        embed = discord.Embed(
            title=f"Application Lookup #{application_id}",
            color=discord.Color.blurple(),
            timestamp=utc_now(),
        )
        embed.add_field(name="Username", value=record.get("username", "N/A"), inline=True)
        embed.add_field(name="User ID", value=str(record.get("user_id", "N/A")), inline=True)
        embed.add_field(name="Status", value=make_status_emoji(record.get("status", "N/A")), inline=True)
        embed.add_field(name="Gamertag", value=record.get("gamer_tag", "N/A"), inline=True)
        embed.add_field(name="Age", value=record.get("age", "N/A"), inline=True)
        embed.add_field(name="Reviewed By", value=record.get("reviewed_by") or "Not reviewed", inline=True)
        embed.add_field(name="Ticket Channel ID", value=str(record.get("ticket_channel_id", "N/A")), inline=False)
        embed.add_field(name="Submitted", value=record.get("submitted_at", "N/A"), inline=False)
        await interaction.response.send_message(embed=embed, ephemeral=True)

    @app_commands.command(name="application_stats", description="See DIFF application totals")
    async def application_stats(self, interaction: discord.Interaction):
        if not isinstance(interaction.user, discord.Member) or not is_staff_reviewer(interaction.user):
            return await interaction.response.send_message(
                "Only Leader, Co-Leader, or Manager can use this command.",
                ephemeral=True,
            )

        data = load_data()
        apps = list(data.get("applications", {}).values())
        pending = sum(1 for a in apps if a.get("status") == "Pending")
        approved = sum(1 for a in apps if a.get("status") == "Approved")
        denied = sum(1 for a in apps if a.get("status") == "Denied")
        total = len(apps)

        embed = discord.Embed(
            title="DIFF Application Stats",
            color=discord.Color.blue(),
            timestamp=utc_now(),
        )
        embed.add_field(name="Total", value=str(total), inline=True)
        embed.add_field(name="Pending", value=str(pending), inline=True)
        embed.add_field(name="Approved", value=str(approved), inline=True)
        embed.add_field(name="Denied", value=str(denied), inline=True)
        await interaction.response.send_message(embed=embed, ephemeral=True)


async def setup(bot: commands.Bot):
    await bot.add_cog(DiffApplicationSystem(bot))
